
-- READ ONLY.
-- Reviews every column for:
-- FND-3528, its shared-view record, and policy DT-RM-001.

WITH SAMPLE_RECORDS AS (

    SELECT
        '1 CURATED FINDING' AS SOURCE_RECORD,
        F.ARCHER_CONTENT_FINDINGS_CONTENT_ID AS RECORD_KEY,
        OBJECT_CONSTRUCT_KEEP_NULL(F.*) AS ROW_DATA
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_FINDINGS F
    WHERE F.ARCHER_CONTENT_FINDINGS_CONTENT_ID = 8391422

    UNION ALL

    SELECT
        '2 SHARED ENTERPRISE_FINDINGS',
        S.ARCHER_CONTENT_FINDINGS_CONTENT_ID,
        OBJECT_CONSTRUCT_KEEP_NULL(S.*)
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.ENTERPRISE_FINDINGS S
    WHERE S.ARCHER_CONTENT_FINDINGS_CONTENT_ID = 8391422

    UNION ALL

    SELECT
        '3 POLICY DT-RM-001',
        P.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3_CONTENT_ID,
        OBJECT_CONSTRUCT_KEEP_NULL(P.*)
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3 P
    WHERE TRIM(P.POLICY_LEVEL_3_NAME) = 'DT-RM-001'
)

SELECT
    R.SOURCE_RECORD,
    R.RECORD_KEY,
    K.KEY::VARCHAR AS FIELD_NAME,
    CASE
        WHEN IS_NULL_VALUE(K.VALUE) THEN '[NULL]'
        WHEN TRIM(K.VALUE::VARCHAR) = '' THEN '[EMPTY]'
        ELSE K.VALUE::VARCHAR
    END AS FIELD_VALUE
FROM SAMPLE_RECORDS R,
     LATERAL FLATTEN(INPUT => R.ROW_DATA) K
ORDER BY
    R.SOURCE_RECORD,
    R.RECORD_KEY,
    FIELD_NAME;
