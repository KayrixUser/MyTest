-- READ ONLY: does not change any table or view.

WITH BASE AS (
    SELECT
        ARCHER_CONTENT_FINDINGS_CONTENT_ID AS FINDING_KEY,
        ARCHER_CONTENT_FINDINGS_CONFIRMED_IN_ARCHER AS CONFIRMED_AT,
        UPPER(TRIM(COALESCE(OVERALL_STATUS, ''))) AS STATUS_VALUE
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_FINDINGS
),

CUTOFF AS (
    -- Table-wide maximum: no business, source, or status filter.
    SELECT MAX(CONFIRMED_AT) AS LATEST_CONFIRMED_AT
    FROM BASE
),

LATEST_CANDIDATE AS (
    SELECT B.*
    FROM BASE B
    CROSS JOIN CUTOFF C
    WHERE B.CONFIRMED_AT = C.LATEST_CONFIRMED_AT
),

POPULATIONS AS (
    SELECT
        '1 - All curated data' AS CHECK_SCOPE,
        FINDING_KEY,
        STATUS_VALUE
    FROM BASE

    UNION ALL

    SELECT
        '2 - Latest-confirmed candidate',
        FINDING_KEY,
        STATUS_VALUE
    FROM LATEST_CANDIDATE

    UNION ALL

    SELECT
        '3 - Existing C2S view',
        FINDING_KEY,
        UPPER(TRIM(COALESCE(OVERALL_STATUS, '')))
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.C2S_ENTERPRISE_FINDINGS_DATA
)

SELECT
    CHECK_SCOPE,
    (SELECT LATEST_CONFIRMED_AT FROM CUTOFF) AS CANDIDATE_CUTOFF,

    COUNT(*) AS ROW_COUNT,
    COUNT(DISTINCT FINDING_KEY) AS DISTINCT_FINDINGS,

    COUNT(*) - COUNT(FINDING_KEY) AS NULL_KEYS,
    COUNT(FINDING_KEY) - COUNT(DISTINCT FINDING_KEY)
        AS DUPLICATE_KEY_ROWS,

    COUNT(DISTINCT CASE
        WHEN STATUS_VALUE = 'OPEN' THEN FINDING_KEY
    END) AS OPEN_FINDINGS,

    COUNT(DISTINCT CASE
        WHEN STATUS_VALUE = 'CLOSED' THEN FINDING_KEY
    END) AS CLOSED_FINDINGS,

    COUNT(DISTINCT CASE
        WHEN STATUS_VALUE = 'DRAFT' THEN FINDING_KEY
    END) AS DRAFT_FINDINGS,

    COUNT(DISTINCT CASE
        WHEN STATUS_VALUE NOT IN ('OPEN', 'CLOSED', 'DRAFT')
        THEN FINDING_KEY
    END) AS OTHER_OR_BLANK_STATUS

FROM POPULATIONS
GROUP BY CHECK_SCOPE
ORDER BY CHECK_SCOPE;
