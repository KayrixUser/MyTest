CREATE OR REPLACE VIEW
    RTX_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.C2S_SOX_FINANCE_TREND_HISTORY
COPY GRANTS
AS
SELECT
    CAPTURE_DATE,
    CAPTURED_AT_UTC,
    FINDING_ID,
    BUSINESS_UNIT_NAME,
    ORGANIZATION_NAME,
    ENTITY_NAME,
    ENTITY,
    ENTITY_TIER,
    CONTROL_LIBRARY,
    CYCLE,
    SUBCYCLE,
    CONTROL_ID,
    REPORTING_YEAR,
    FISCAL_YEAR,
    FINDING_TITLE,
    FINDING,
    OVERALL_STATUS,
    FINDING_SOURCE,
    FINDING_TYPE,
    CRITICALITY,
    RECORD_CREATOR,
    ASSIGNED_TO,
    REVIEWER,
    SUBMIT_DATE,
    REVIEW_DATE,
    COMMITTED_CLOSURE_DATE,
    REVISED_CLOSURE_DATE,
    DATE_CREATED,
    DATE_CLOSED,
    KEY_CONTROL,
    RTX_SOX_SCOPE,
    TRACKING_ID,
    INSERT_DATE,
    UPDATE_DATE,
    CHECKSUM_VALUE,

    DATEDIFF(
        'day', COMMITTED_CLOSURE_DATE, CAPTURE_DATE
    ) AS DAYS_PAST_DUE,

    CASE
        WHEN CAPTURE_DATE IS NULL
          OR COMMITTED_CLOSURE_DATE IS NULL
            THEN NULL

        WHEN DATEDIFF(
                 'day', COMMITTED_CLOSURE_DATE, CAPTURE_DATE
             ) <= 30
            THEN 'Target Met'

        WHEN DATEDIFF(
                 'day', COMMITTED_CLOSURE_DATE, CAPTURE_DATE
             ) > 30
         AND REVISED_CLOSURE_DATE >= CAPTURE_DATE
         AND YEAR(REVISED_CLOSURE_DATE) = YEAR(CAPTURE_DATE)
            THEN 'Target at Risk'

        WHEN DATEDIFF(
                 'day', COMMITTED_CLOSURE_DATE, CAPTURE_DATE
             ) > 30
         AND (
                REVISED_CLOSURE_DATE IS NULL
                OR REVISED_CLOSURE_DATE < CAPTURE_DATE
                OR YEAR(REVISED_CLOSURE_DATE) > YEAR(CAPTURE_DATE)
             )
            THEN 'Target Unmet'

        ELSE NULL
    END AS TARGET_STATUS

FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.C2S_SOX_FINANCE_TREND_HISTORY
WHERE LOWER(TRIM(OVERALL_STATUS)) = 'in process';


--Test
SELECT
    CAPTURE_DATE,
    CAPTURED_AT_UTC,
    BUSINESS_UNIT_NAME,
    COUNT(*) AS FINDING_ROWS,
    COUNT(DISTINCT FINDING_ID) AS DISTINCT_FINDINGS,
    SUM(CASE WHEN TARGET_STATUS = 'Target Met'
             THEN 1 ELSE 0 END) AS TARGET_MET,
    SUM(CASE WHEN TARGET_STATUS = 'Target at Risk'
             THEN 1 ELSE 0 END) AS TARGET_AT_RISK,
    SUM(CASE WHEN TARGET_STATUS = 'Target Unmet'
             THEN 1 ELSE 0 END) AS TARGET_UNMET,
    COUNT(*) - COUNT(TARGET_STATUS) AS UNCLASSIFIED
FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.C2S_SOX_FINANCE_TREND_HISTORY
WHERE CAPTURE_DATE = DATE '2026-10-26'
GROUP BY CAPTURE_DATE, CAPTURED_AT_UTC, BUSINESS_UNIT_NAME
ORDER BY BUSINESS_UNIT_NAME;
