
-- READ ONLY: compares the same Finding across three existing tables.
-- Does not modify data, views, or Matillion jobs.

WITH SOURCE_RECORDS AS (
    SELECT
        '1 STAGING' AS DATA_LAYER,
        OBJECT_CONSTRUCT_KEEP_NULL(F.*) AS ROW_DATA
    FROM RTX_RAW.ES_ESC_GRC.ARCHER_CONTENT_FINDINGS_STG F
    WHERE F.ARCHER_CONTENT_FINDINGS_CONTENT_ID = 8391422

    UNION ALL

    SELECT
        '2 RAW' AS DATA_LAYER,
        OBJECT_CONSTRUCT_KEEP_NULL(F.*) AS ROW_DATA
    FROM RTX_RAW.ES_ESC_GRC.ARCHER_CONTENT_FINDINGS F
    WHERE F.ARCHER_CONTENT_FINDINGS_CONTENT_ID = 8391422

    UNION ALL

    SELECT
        '3 CURATED' AS DATA_LAYER,
        OBJECT_CONSTRUCT_KEEP_NULL(F.*) AS ROW_DATA
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_FINDINGS F
    WHERE F.ARCHER_CONTENT_FINDINGS_CONTENT_ID = 8391422
),
LAYERS AS (
    SELECT COLUMN1 AS DATA_LAYER
    FROM VALUES ('1 STAGING'), ('2 RAW'), ('3 CURATED')
),
CHECK_VALUES AS (
    SELECT
        L.DATA_LAYER,
        COUNT(S.DATA_LAYER) OVER (
            PARTITION BY L.DATA_LAYER
        ) AS MATCHING_ROWS,
        GET(S.ROW_DATA, 'POLICY_LEVEL_3') AS SINGLE_UNDERSCORE_VALUE,
        GET(S.ROW_DATA, 'POLICY__LEVEL_3') AS DOUBLE_UNDERSCORE_VALUE,
        GET(S.ROW_DATA, 'UPDATE_DATE')::VARCHAR AS LOAD_UPDATE_DATE,
        GET(
            S.ROW_DATA,
            'ARCHER_CONTENT_FINDINGS_LAST_UPDATED'
        )::VARCHAR AS ARCHER_LAST_UPDATED
    FROM LAYERS L
    LEFT JOIN SOURCE_RECORDS S
        ON L.DATA_LAYER = S.DATA_LAYER
)
SELECT
    DATA_LAYER,
    MATCHING_ROWS,

    CASE
        WHEN MATCHING_ROWS = 0 THEN '[RECORD NOT FOUND]'
        WHEN SINGLE_UNDERSCORE_VALUE IS NULL THEN '[COLUMN ABSENT]'
        WHEN IS_NULL_VALUE(SINGLE_UNDERSCORE_VALUE) THEN '[NULL]'
        WHEN TRIM(SINGLE_UNDERSCORE_VALUE::VARCHAR) = '' THEN '[EMPTY]'
        ELSE SINGLE_UNDERSCORE_VALUE::VARCHAR
    END AS POLICY_LEVEL_3,

    CASE
        WHEN MATCHING_ROWS = 0 THEN '[RECORD NOT FOUND]'
        WHEN DOUBLE_UNDERSCORE_VALUE IS NULL THEN '[COLUMN ABSENT]'
        WHEN IS_NULL_VALUE(DOUBLE_UNDERSCORE_VALUE) THEN '[NULL]'
        WHEN TRIM(DOUBLE_UNDERSCORE_VALUE::VARCHAR) = '' THEN '[EMPTY]'
        ELSE DOUBLE_UNDERSCORE_VALUE::VARCHAR
    END AS POLICY__LEVEL_3,

    LOAD_UPDATE_DATE,
    ARCHER_LAST_UPDATED
FROM CHECK_VALUES
ORDER BY DATA_LAYER;
