
-- READ ONLY: no table, view, or Power BI changes.
SELECT
    ARCHER_CONTENT_FINDINGS_CONFIRMED_IN_ARCHER AS CONFIRMED_AT,
    SOURCE AS FINDING_SOURCE,
    COUNT(*) AS SOURCE_ROWS,
    COUNT(DISTINCT ARCHER_CONTENT_FINDINGS_CONTENT_ID)
        AS DISTINCT_FINDINGS,

    COUNT(CASE
        WHEN ARCHER_CONTENT_FINDINGS_CONTENT_ID IS NULL THEN 1
    END) AS NULL_FINDING_KEYS,

    COUNT(DISTINCT CASE
        WHEN UPPER(TRIM(OVERALL_STATUS)) = 'OPEN'
        THEN ARCHER_CONTENT_FINDINGS_CONTENT_ID
    END) AS OPEN_FINDINGS,

    COUNT(DISTINCT CASE
        WHEN UPPER(TRIM(OVERALL_STATUS)) = 'CLOSED'
        THEN ARCHER_CONTENT_FINDINGS_CONTENT_ID
    END) AS CLOSED_FINDINGS,

    COUNT(DISTINCT CASE
        WHEN UPPER(TRIM(OVERALL_STATUS)) = 'DRAFT'
        THEN ARCHER_CONTENT_FINDINGS_CONTENT_ID
    END) AS DRAFT_FINDINGS

FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_FINDINGS

GROUP BY
    ARCHER_CONTENT_FINDINGS_CONFIRMED_IN_ARCHER,
    SOURCE

ORDER BY
    CONFIRMED_AT DESC NULLS LAST,
    FINDING_SOURCE;
