
WITH FINDING_VALUES AS (
    SELECT
        K.KEY::VARCHAR AS SQL_COLUMN,
        K.VALUE AS FIELD_VALUE
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_FINDINGS F,
         LATERAL FLATTEN(
             INPUT => OBJECT_CONSTRUCT_KEEP_NULL(F.*)
         ) K
    WHERE F.ARCHER_CONTENT_FINDINGS_CONTENT_ID = 8391422
)

SELECT DISTINCT
    M.FIELDID,
    M.FIELDNAME,
    M.SQLFIELDNAME,
    M.FIELDTYPE,
    M.REFLEVELID,
    M.REFFIELDID,

    CASE
        WHEN V.SQL_COLUMN IS NULL
            THEN '[NO MATCHING COLUMN IN RECORD]'
        WHEN IS_NULL_VALUE(V.FIELD_VALUE)
            THEN '[NULL]'
        WHEN TRIM(V.FIELD_VALUE::VARCHAR) = ''
            THEN '[EMPTY]'
        ELSE V.FIELD_VALUE::VARCHAR
    END AS FND_3528_CURATED_VALUE,

    M.DESCRIPTION

FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_META_FIELDS M

JOIN RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_META_LEVELS L
    ON M.LVLID = L.LVLID

LEFT JOIN FINDING_VALUES V
    ON UPPER(TRIM(M.SQLFIELDNAME)) = UPPER(V.SQL_COLUMN)

WHERE (
       UPPER(TRIM(L.LVLNAME)) = 'FINDINGS'
    OR UPPER(TRIM(L.SQLTABLENAME)) = 'ARCHER_CONTENT_FINDINGS'
)
AND (
       M.FIELDNAME ILIKE '%POLIC%'
    OR M.SQLFIELDNAME ILIKE '%POLIC%'
)

ORDER BY M.FIELDNAME, M.FIELDID;
