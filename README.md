    COALESCE(COUNT_IF(NOT EQUAL_NULL(P.ADDITIONAL_READ_ACCESS, F.ADDITIONAL_READ_ACCESS)), 0)
        AS ADDITIONAL_READ_ACCESS_MISMATCHES
FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.ENTERPRISE_FINDINGS P
JOIN RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_FINDINGS F
    ON P.ARCHER_CONTENT_FINDINGS_CONTENT_ID = F.ARCHER_CONTENT_FINDINGS_CONTENT_ID;
