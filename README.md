-- READ ONLY: does not change any table, view, or Power BI model.
-- Replace only PASTE_PREVIEW_QUERY_ID.

WITH PREVIEW_OUTPUT AS (
    SELECT *
    FROM TABLE(RESULT_SCAN('PASTE_PREVIEW_QUERY_ID'))
),

RAW_RP AS (
    SELECT
        ARCHER_CONTENT_REMEDIATION_PLANS_CONTENT_ID AS PLAN_KEY,
        REMEDIATION_PLAN_ID AS PLAN_NUMBER,
        STATUS AS PLAN_STATUS,
        ARCHER_CONTENT_REMEDIATION_PLANS_CONFIRMED_IN_ARCHER
            AS CONFIRMED_AT
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_REMEDIATION_PLANS
),

CUTOFF AS (
    SELECT MAX(CONFIRMED_AT) AS LATEST_CONFIRMED_AT
    FROM RAW_RP
),

LATEST_RP AS (
    SELECT R.*
    FROM RAW_RP R
    CROSS JOIN CUTOFF C
    WHERE R.CONFIRMED_AT = C.LATEST_CONFIRMED_AT
),

LATEST_KEYS AS (
    SELECT DISTINCT PLAN_KEY
    FROM LATEST_RP
),

SHARED_COUNTS AS (
    SELECT
        P.ARCHER_CONTENT_REMEDIATION_PLANS_CONTENT_ID AS PLAN_KEY,
        COUNT(*) AS ROWS_PER_PLAN
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.ENTERPRISE_REMEDIATION_PLANS P
    WHERE EXISTS (
        SELECT 1
        FROM LATEST_KEYS K
        WHERE K.PLAN_KEY =
              P.ARCHER_CONTENT_REMEDIATION_PLANS_CONTENT_ID
    )
    GROUP BY P.ARCHER_CONTENT_REMEDIATION_PLANS_CONTENT_ID
),

EXPECTED_OUTPUT AS (
    SELECT V.*
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.C2S_ENTERPRISE_REMEDIATION_PLANS_DATA V
    WHERE EXISTS (
        SELECT 1
        FROM LATEST_KEYS K
        WHERE K.PLAN_KEY = V.REMEDIATION_PLAN_KEY
    )
),

-- Compare every output value AND each complete row's occurrence count.
EXPECTED_ROWS AS (
    SELECT E.*, COUNT(*) AS QA_ROW_COPIES
    FROM EXPECTED_OUTPUT E
    GROUP BY ALL
),

PREVIEW_ROWS AS (
    SELECT P.*, COUNT(*) AS QA_ROW_COPIES
    FROM PREVIEW_OUTPUT P
    GROUP BY ALL
),

MISSING_KEYS AS (
    SELECT PLAN_KEY FROM LATEST_KEYS
    EXCEPT
    SELECT REMEDIATION_PLAN_KEY FROM PREVIEW_OUTPUT
),

EXTRA_KEYS AS (
    SELECT REMEDIATION_PLAN_KEY FROM PREVIEW_OUTPUT
    EXCEPT
    SELECT PLAN_KEY FROM LATEST_KEYS
),

MISSING_OR_CHANGED_ROWS AS (
    SELECT * FROM EXPECTED_ROWS
    EXCEPT
    SELECT * FROM PREVIEW_ROWS
),

EXTRA_OR_CHANGED_ROWS AS (
    SELECT * FROM PREVIEW_ROWS
    EXCEPT
    SELECT * FROM EXPECTED_ROWS
),

CHECKS AS (
    SELECT '01 Latest source distinct plans' AS CHECK_NAME,
           COUNT(DISTINCT PLAN_KEY) AS RESULT
    FROM LATEST_RP

    UNION ALL
    SELECT '02 Preview distinct plans',
           COUNT(DISTINCT REMEDIATION_PLAN_KEY)
    FROM PREVIEW_OUTPUT

    UNION ALL
    SELECT '03 Existing-view rows for eligible plans', COUNT(*)
    FROM EXPECTED_OUTPUT

    UNION ALL
    SELECT '04 Preview membership rows', COUNT(*)
    FROM PREVIEW_OUTPUT

    UNION ALL
    SELECT '05 Latest source null keys',
           COUNT(*) - COUNT(PLAN_KEY)
    FROM LATEST_RP

    UNION ALL
    SELECT '06 Latest source duplicate-key rows',
           COUNT(PLAN_KEY) - COUNT(DISTINCT PLAN_KEY)
    FROM LATEST_RP

    UNION ALL
    SELECT '07 Latest plans missing from shared published view', COUNT(*)
    FROM LATEST_KEYS K
    WHERE NOT EXISTS (
        SELECT 1
        FROM SHARED_COUNTS S
        WHERE S.PLAN_KEY = K.PLAN_KEY
    )

    UNION ALL
    SELECT '08 Duplicate plan rows in shared published input',
           COALESCE(SUM(ROWS_PER_PLAN - 1), 0)
    FROM SHARED_COUNTS

    UNION ALL
    SELECT '09 Preview null plan keys',
           COUNT(*) - COUNT(REMEDIATION_PLAN_KEY)
    FROM PREVIEW_OUTPUT

    UNION ALL
    SELECT '10 Eligible plan keys missing from preview', COUNT(*)
    FROM MISSING_KEYS

    UNION ALL
    SELECT '11 Preview keys outside latest population', COUNT(*)
    FROM EXTRA_KEYS

    UNION ALL
    SELECT '12 Expected rows missing or changed including copies', COUNT(*)
    FROM MISSING_OR_CHANGED_ROWS

    UNION ALL
    SELECT '13 Preview rows extra or changed including copies', COUNT(*)
    FROM EXTRA_OR_CHANGED_ROWS

    UNION ALL
    SELECT '14 Plan statuses differing from latest curated source',
           COUNT(DISTINCT P.REMEDIATION_PLAN_KEY)
    FROM PREVIEW_OUTPUT P
    JOIN LATEST_RP R
      ON R.PLAN_KEY = P.REMEDIATION_PLAN_KEY
    WHERE P.STATUS IS DISTINCT FROM R.PLAN_STATUS

    UNION ALL
    SELECT '15 RP-2115 in latest source population',
           COUNT(DISTINCT PLAN_KEY)
    FROM LATEST_RP
    WHERE PLAN_NUMBER = 2115

    UNION ALL
    SELECT '16 RP-2115 in preview',
           COUNT(DISTINCT P.REMEDIATION_PLAN_KEY)
    FROM PREVIEW_OUTPUT P
    WHERE EXISTS (
        SELECT 1
        FROM RAW_RP R
        WHERE R.PLAN_KEY = P.REMEDIATION_PLAN_KEY
          AND R.PLAN_NUMBER = 2115
    )
)

SELECT
    C.LATEST_CONFIRMED_AT AS CURRENT_SOURCE_CUTOFF,
    T.CHECK_NAME,
    T.RESULT
FROM CHECKS T
CROSS JOIN CUTOFF C
ORDER BY T.CHECK_NAME;
