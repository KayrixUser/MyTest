
-- SECTION 1: Replace the existing dashboard-specific mapping.
CREATE OR REPLACE VIEW RTX_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.C2S_ENTERPRISE_FINDINGS_POLICY_DATA
(
    FINDING_KEY,
    POLICY_LEVEL_3_KEY,
    POLICY_LEVEL_3_REFERENCE,
    POLICY_LEVEL_3_TITLE
)
COPY GRANTS
AS
WITH OPEN_FINDINGS AS (
    SELECT
        ARCHER_CONTENT_FINDINGS_CONTENT_ID AS FINDING_KEY,
        CONTROL_STANDARDS
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.ENTERPRISE_FINDINGS
    WHERE OVERALL_STATUS = 'Open'
),
CONTROL_REFERENCES AS (
    SELECT DISTINCT
        F.FINDING_KEY,
        NULLIF(TRIM(S.VALUE), '') AS CONTROL_ID_RAW
    FROM OPEN_FINDINGS F,
         LATERAL SPLIT_TO_TABLE(
             REPLACE(COALESCE(TO_VARCHAR(F.CONTROL_STANDARDS), ''), ',', ';'),
             ';'
         ) S
),
CONTROL_LINKS AS (
    SELECT
        R.FINDING_KEY,
        R.CONTROL_ID_RAW,
        C.ARCHER_CONTENT_CONTROL_STANDARDS_CONTENT_ID AS MATCHED_CONTROL_KEY,
        C.POLICY_LEVEL_3_NAME AS POLICY_IDS_RAW
    FROM CONTROL_REFERENCES R
    LEFT JOIN RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_CONTROL_STANDARDS C
        ON TRY_TO_NUMBER(R.CONTROL_ID_RAW)
           = C.ARCHER_CONTENT_CONTROL_STANDARDS_CONTENT_ID
),
POLICY_REFERENCES AS (
    SELECT
        C.FINDING_KEY,
        C.CONTROL_ID_RAW,
        C.MATCHED_CONTROL_KEY,
        NULLIF(TRIM(S.VALUE), '') AS POLICY_ID_RAW
    FROM CONTROL_LINKS C,
         LATERAL SPLIT_TO_TABLE(
             REPLACE(COALESCE(TO_VARCHAR(C.POLICY_IDS_RAW), ''), ',', ';'),
             ';'
         ) S
),
RESOLVED AS (
    SELECT
        R.*,
        P.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3_CONTENT_ID AS MATCHED_POLICY_KEY,
        NULLIF(TRIM(P.POLICY_LEVEL_3_NAME), '') AS POLICY_REFERENCE,
        NULLIF(TRIM(P.POLICY_NAME), '') AS POLICY_TITLE
    FROM POLICY_REFERENCES R
    LEFT JOIN RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3 P
        ON TRY_TO_NUMBER(R.POLICY_ID_RAW)
           = P.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3_CONTENT_ID
)
SELECT DISTINCT
    FINDING_KEY,
    MATCHED_POLICY_KEY AS POLICY_LEVEL_3_KEY,
    POLICY_REFERENCE AS POLICY_LEVEL_3_REFERENCE,
    POLICY_TITLE AS POLICY_LEVEL_3_TITLE
FROM RESOLVED
WHERE FINDING_KEY IS NOT NULL
  AND MATCHED_CONTROL_KEY IS NOT NULL
  AND MATCHED_POLICY_KEY IS NOT NULL
  AND POLICY_REFERENCE IS NOT NULL;

-- SECTION 2: Read-only post-change check against the SAME source route.
-- Execute this entire WITH ... SELECT together when rerunning it separately.
WITH OPEN_FINDINGS AS (
    SELECT
        ARCHER_CONTENT_FINDINGS_CONTENT_ID AS FINDING_KEY,
        CONTROL_STANDARDS
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.ENTERPRISE_FINDINGS
    WHERE OVERALL_STATUS = 'Open'
),
CONTROL_REFERENCES AS (
    SELECT DISTINCT
        F.FINDING_KEY,
        NULLIF(TRIM(S.VALUE), '') AS CONTROL_ID_RAW
    FROM OPEN_FINDINGS F,
         LATERAL SPLIT_TO_TABLE(
             REPLACE(COALESCE(TO_VARCHAR(F.CONTROL_STANDARDS), ''), ',', ';'),
             ';'
         ) S
),
CONTROL_LINKS AS (
    SELECT
        R.FINDING_KEY,
        R.CONTROL_ID_RAW,
        C.ARCHER_CONTENT_CONTROL_STANDARDS_CONTENT_ID AS MATCHED_CONTROL_KEY,
        C.POLICY_LEVEL_3_NAME AS POLICY_IDS_RAW
    FROM CONTROL_REFERENCES R
    LEFT JOIN RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_CONTROL_STANDARDS C
        ON TRY_TO_NUMBER(R.CONTROL_ID_RAW)
           = C.ARCHER_CONTENT_CONTROL_STANDARDS_CONTENT_ID
),
POLICY_REFERENCES AS (
    SELECT
        C.FINDING_KEY,
        C.CONTROL_ID_RAW,
        C.MATCHED_CONTROL_KEY,
        NULLIF(TRIM(S.VALUE), '') AS POLICY_ID_RAW
    FROM CONTROL_LINKS C,
         LATERAL SPLIT_TO_TABLE(
             REPLACE(COALESCE(TO_VARCHAR(C.POLICY_IDS_RAW), ''), ',', ';'),
             ';'
         ) S
),
RESOLVED AS (
    SELECT
        R.*,
        P.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3_CONTENT_ID AS MATCHED_POLICY_KEY,
        NULLIF(TRIM(P.POLICY_LEVEL_3_NAME), '') AS POLICY_REFERENCE,
        NULLIF(TRIM(P.POLICY_NAME), '') AS POLICY_TITLE
    FROM POLICY_REFERENCES R
    LEFT JOIN RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3 P
        ON TRY_TO_NUMBER(R.POLICY_ID_RAW)
           = P.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3_CONTENT_ID
),
EXPECTED_ROWS AS (
SELECT DISTINCT
    FINDING_KEY,
    MATCHED_POLICY_KEY AS POLICY_LEVEL_3_KEY,
    POLICY_REFERENCE AS POLICY_LEVEL_3_REFERENCE,
    POLICY_TITLE AS POLICY_LEVEL_3_TITLE
FROM RESOLVED
WHERE FINDING_KEY IS NOT NULL
  AND MATCHED_CONTROL_KEY IS NOT NULL
  AND MATCHED_POLICY_KEY IS NOT NULL
  AND POLICY_REFERENCE IS NOT NULL
),
SAVED_ROWS AS (
    SELECT
        FINDING_KEY,
        POLICY_LEVEL_3_KEY,
        POLICY_LEVEL_3_REFERENCE,
        POLICY_LEVEL_3_TITLE
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.C2S_ENTERPRISE_FINDINGS_POLICY_DATA
),
MISSING_OR_CHANGED AS (
    SELECT * FROM EXPECTED_ROWS
    MINUS
    SELECT * FROM SAVED_ROWS
),
EXTRA_OR_CHANGED AS (
    SELECT * FROM SAVED_ROWS
    MINUS
    SELECT * FROM EXPECTED_ROWS
),
DUPLICATE_PAIRS AS (
    SELECT FINDING_KEY, POLICY_LEVEL_3_KEY
    FROM SAVED_ROWS
    GROUP BY FINDING_KEY, POLICY_LEVEL_3_KEY
    HAVING COUNT(*) > 1
)
SELECT '01 Current expected Finding-Policy rows' AS CHECK_NAME,
       COUNT(*) AS RESULT
FROM EXPECTED_ROWS
UNION ALL
SELECT '02 Saved mapping rows', COUNT(*)
FROM SAVED_ROWS
UNION ALL
SELECT '03 Saved distinct Finding-Policy pairs',
       COUNT(DISTINCT FINDING_KEY, POLICY_LEVEL_3_KEY)
FROM SAVED_ROWS
UNION ALL
SELECT '04 Saved distinct mapped Open Findings', COUNT(DISTINCT FINDING_KEY)
FROM SAVED_ROWS
UNION ALL
SELECT '05 Missing or changed source-to-view rows', COUNT(*)
FROM MISSING_OR_CHANGED
UNION ALL
SELECT '06 Extra or changed source-to-view rows', COUNT(*)
FROM EXTRA_OR_CHANGED
UNION ALL
SELECT '07 Duplicate Finding-Policy pairs', COUNT(*)
FROM DUPLICATE_PAIRS
UNION ALL
SELECT '08 Saved rows with null keys or missing policy code', COUNT(*)
FROM SAVED_ROWS
WHERE FINDING_KEY IS NULL
   OR POLICY_LEVEL_3_KEY IS NULL
   OR NULLIF(TRIM(POLICY_LEVEL_3_REFERENCE), '') IS NULL
UNION ALL
SELECT '09 Mapped Finding keys absent from dashboard Open Findings',
       COUNT(DISTINCT V.FINDING_KEY)
FROM SAVED_ROWS V
WHERE NOT EXISTS (
    SELECT 1
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.C2S_ENTERPRISE_FINDINGS_DATA F
    WHERE F.FINDING_KEY = V.FINDING_KEY
      AND F.OVERALL_STATUS = 'Open'
)
UNION ALL
SELECT '10 Current shared-view distinct Open Findings',
       COUNT(DISTINCT FINDING_KEY)
FROM OPEN_FINDINGS
UNION ALL
SELECT '11 Shared Open Findings without a saved policy-code link',
       COUNT(DISTINCT F.FINDING_KEY)
FROM OPEN_FINDINGS F
WHERE NOT EXISTS (
    SELECT 1 FROM SAVED_ROWS V WHERE V.FINDING_KEY = F.FINDING_KEY
)
UNION ALL
SELECT '12 Distinct mapped policies with missing titles - informational',
       COUNT(DISTINCT POLICY_LEVEL_3_KEY)
FROM SAVED_ROWS
WHERE NULLIF(TRIM(POLICY_LEVEL_3_TITLE), '') IS NULL
ORDER BY CHECK_NAME;
