CREATE OR REPLACE VIEW
RTX_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.C2S_ENTERPRISE_FINDINGS_POLICY_DATA
(
    FINDING_KEY,
    POLICY_LEVEL_3_KEY,
    POLICY_LEVEL_3_REFERENCE,
    POLICY_LEVEL_3_TITLE
)
COPY GRANTS
AS
WITH OPEN_FINDINGS AS (
    SELECT
        FINDING_KEY,
        TO_VARCHAR(POLICIES_LEVEL_3) AS POLICY_IDS_RAW
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.C2S_ENTERPRISE_FINDINGS_DATA
    WHERE OVERALL_STATUS = 'Open'
),
TOKENS AS (
    SELECT
        F.FINDING_KEY,
        TRIM(S.VALUE) AS POLICY_ID_RAW
    FROM OPEN_FINDINGS F,
         LATERAL SPLIT_TO_TABLE(
             COALESCE(F.POLICY_IDS_RAW, ''),
             ';'
         ) S
    WHERE NULLIF(TRIM(S.VALUE), '') IS NOT NULL
),
PARSED AS (
    SELECT
        FINDING_KEY,
        CASE
            WHEN REGEXP_LIKE(POLICY_ID_RAW, '^[0-9]+$')
            THEN TRY_TO_NUMBER(POLICY_ID_RAW)
        END AS POLICY_KEY
    FROM TOKENS
),
POLICY_PAIRS AS (
    SELECT DISTINCT
        FINDING_KEY,
        POLICY_KEY
    FROM PARSED
    WHERE FINDING_KEY IS NOT NULL
      AND POLICY_KEY IS NOT NULL
)
SELECT DISTINCT
    R.FINDING_KEY,
    P.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3_CONTENT_ID
        AS POLICY_LEVEL_3_KEY,
    NULLIF(TRIM(P.POLICY_LEVEL_3_NAME), '')
        AS POLICY_LEVEL_3_REFERENCE,
    NULLIF(TRIM(P.POLICY_NAME), '')
        AS POLICY_LEVEL_3_TITLE
FROM POLICY_PAIRS R
JOIN RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3 P
    ON P.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3_CONTENT_ID = R.POLICY_KEY
WHERE NULLIF(TRIM(P.POLICY_LEVEL_3_NAME), '') IS NOT NULL;





SELECT
    COUNT(*) AS PAIR_ROWS,
    COUNT(DISTINCT FINDING_KEY, POLICY_LEVEL_3_KEY) AS DISTINCT_PAIRS,
    COUNT(DISTINCT FINDING_KEY) AS LINKED_OPEN_FINDINGS,
    COUNT(DISTINCT POLICY_LEVEL_3_KEY) AS DISTINCT_POLICIES,
    COUNT(DISTINCT POLICY_LEVEL_3_REFERENCE) AS DISTINCT_REFERENCES,
    COUNT(POLICY_LEVEL_3_TITLE) AS POPULATED_TITLES
FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.C2S_ENTERPRISE_FINDINGS_POLICY_DATA;
