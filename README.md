;WITH Checks AS (
    SELECT
        COUNT_BIG(*) AS TotalRows,

        COUNT_BIG(CASE WHEN HOSTNAME IS NULL THEN 1 END)
            AS Null_HOSTNAME,
        COUNT_BIG(CASE WHEN FQDN IS NULL THEN 1 END)
            AS Null_FQDN,
        COUNT_BIG(CASE WHEN BUSINESS_UNIT IS NULL THEN 1 END)
            AS Null_BUSINESS_UNIT,
        COUNT_BIG(CASE WHEN DESCRIPTION IS NULL THEN 1 END)
            AS Null_DESCRIPTION,
        COUNT_BIG(CASE WHEN FIRST_SEEN_UTC IS NULL THEN 1 END)
            AS Null_FIRST_SEEN_UTC,
        COUNT_BIG(CASE WHEN RISK_SCORE IS NULL THEN 1 END)
            AS Null_RISK_SCORE,
        COUNT_BIG(CASE WHEN SEVERITY IS NULL THEN 1 END)
            AS Null_SEVERITY,
        COUNT_BIG(CASE WHEN IS_EXTERNAL IS NULL THEN 1 END)
            AS Null_IS_EXTERNAL,
        COUNT_BIG(CASE WHEN TITLE IS NULL THEN 1 END)
            AS Null_TITLE,

        COUNT_BIG(CASE WHEN DATALENGTH(SAPS) > 8000
            THEN 1 END) AS SAPS_Exceeds4000,

        COUNT_BIG(CASE WHEN
            DATALENGTH(CONVERT(varchar(max), SEVERITY)) > 64
            THEN 1 END) AS SEVERITY_Exceeds64Bytes,

        COUNT_BIG(CASE WHEN RISK_SCORE IS NOT NULL
            AND TRY_CONVERT(int, RISK_SCORE) IS NULL
            THEN 1 END) AS RiskScore_OutsideInt,

        COUNT_BIG(CASE WHEN FIRST_SEEN_UTC IS NOT NULL
            AND TRY_CONVERT(datetime, FIRST_SEEN_UTC) IS NULL
            THEN 1 END) AS FirstSeen_OutsideDatetime,

        COUNT_BIG(CASE WHEN
            CONVERT(datetime2(3),
                TRY_CONVERT(datetime, FIRST_SEEN_UTC))
                <> FIRST_SEEN_UTC
            THEN 1 END) AS FirstSeen_RoundingChanges,

        COUNT_BIG(CASE WHEN CONVERT(varbinary(max), HOSTNAME) <>
            CONVERT(varbinary(max), CONVERT(nvarchar(max),
                CONVERT(varchar(max), HOSTNAME)))
            THEN 1 END) AS HOSTNAME_VarcharChanges,

        COUNT_BIG(CASE WHEN CONVERT(varbinary(max), FQDN) <>
            CONVERT(varbinary(max), CONVERT(nvarchar(max),
                CONVERT(varchar(max), FQDN)))
            THEN 1 END) AS FQDN_VarcharChanges,

        COUNT_BIG(CASE WHEN CONVERT(varbinary(max), BUSINESS_UNIT) <>
            CONVERT(varbinary(max), CONVERT(nvarchar(max),
                CONVERT(varchar(max), BUSINESS_UNIT)))
            THEN 1 END) AS BUSINESS_UNIT_VarcharChanges,

        COUNT_BIG(CASE WHEN CONVERT(varbinary(max), SEVERITY) <>
            CONVERT(varbinary(max), CONVERT(nvarchar(max),
                CONVERT(varchar(max), SEVERITY)))
            THEN 1 END) AS SEVERITY_VarcharChanges
    FROM ACP_Connection.dbo.C2S_TLS_ASSETS_DATA
)
SELECT CheckName, RowsFound
FROM Checks
UNPIVOT (
    RowsFound FOR CheckName IN (
        TotalRows,
        Null_HOSTNAME,
        Null_FQDN,
        Null_BUSINESS_UNIT,
        Null_DESCRIPTION,
        Null_FIRST_SEEN_UTC,
        Null_RISK_SCORE,
        Null_SEVERITY,
        Null_IS_EXTERNAL,
        Null_TITLE,
        SAPS_Exceeds4000,
        SEVERITY_Exceeds64Bytes,
        RiskScore_OutsideInt,
        FirstSeen_OutsideDatetime,
        FirstSeen_RoundingChanges,
        HOSTNAME_VarcharChanges,
        FQDN_VarcharChanges,
        BUSINESS_UNIT_VarcharChanges,
        SEVERITY_VarcharChanges
    )
) U
ORDER BY CASE WHEN CheckName = 'TotalRows' THEN 0 ELSE 1 END,
         CheckName;
