-- READ ONLY. No table, view, or Power BI changes.
-- Replace only PASTE_PREVIEW_QUERY_ID with the preview SELECT's Query ID.

WITH PREVIEW_OUTPUT AS (
    SELECT *
    FROM TABLE(RESULT_SCAN('PASTE_PREVIEW_QUERY_ID'))
),

LATEST AS (
    SELECT
        MAX(ARCHER_CONTENT_FINDINGS_CONFIRMED_IN_ARCHER) AS CUTOFF
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_FINDINGS
),

ELIGIBLE_KEYS AS (
    SELECT
        F.ARCHER_CONTENT_FINDINGS_CONTENT_ID AS FINDING_KEY
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_FINDINGS F
    CROSS JOIN LATEST L
    WHERE F.ARCHER_CONTENT_FINDINGS_CONFIRMED_IN_ARCHER = L.CUTOFF
),

EXPECTED_OUTPUT AS (
    SELECT V.*
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.C2S_ENTERPRISE_FINDINGS_DATA V
    WHERE EXISTS (
        SELECT 1
        FROM ELIGIBLE_KEYS K
        WHERE K.FINDING_KEY = V.FINDING_KEY
    )
),

MISSING_KEYS AS (
    SELECT FINDING_KEY FROM ELIGIBLE_KEYS
    EXCEPT
    SELECT FINDING_KEY FROM PREVIEW_OUTPUT
),

EXTRA_KEYS AS (
    SELECT FINDING_KEY FROM PREVIEW_OUTPUT
    EXCEPT
    SELECT FINDING_KEY FROM ELIGIBLE_KEYS
),

MISSING_OR_CHANGED_ROWS AS (
    SELECT * FROM EXPECTED_OUTPUT
    EXCEPT
    SELECT * FROM PREVIEW_OUTPUT
),

EXTRA_OR_CHANGED_ROWS AS (
    SELECT * FROM PREVIEW_OUTPUT
    EXCEPT
    SELECT * FROM EXPECTED_OUTPUT
),

CHECKS AS (
    SELECT '01 Latest source distinct keys' AS CHECK_NAME,
           COUNT(DISTINCT FINDING_KEY) AS RESULT
    FROM ELIGIBLE_KEYS

    UNION ALL
    SELECT '02 Existing-view rows for eligible keys', COUNT(*)
    FROM EXPECTED_OUTPUT

    UNION ALL
    SELECT '03 Preview rows', COUNT(*)
    FROM PREVIEW_OUTPUT

    UNION ALL
    SELECT '04 Preview distinct keys', COUNT(DISTINCT FINDING_KEY)
    FROM PREVIEW_OUTPUT

    UNION ALL
    SELECT '05 Preview null keys', COUNT(*) - COUNT(FINDING_KEY)
    FROM PREVIEW_OUTPUT

    UNION ALL
    SELECT '06 Preview duplicate-key rows',
           COUNT(FINDING_KEY) - COUNT(DISTINCT FINDING_KEY)
    FROM PREVIEW_OUTPUT

    UNION ALL
    SELECT '07 Eligible keys missing from preview', COUNT(*)
    FROM MISSING_KEYS

    UNION ALL
    SELECT '08 Preview keys outside latest population', COUNT(*)
    FROM EXTRA_KEYS

    UNION ALL
    SELECT '09 Expected rows missing or changed', COUNT(*)
    FROM MISSING_OR_CHANGED_ROWS

    UNION ALL
    SELECT '10 Preview rows extra or changed', COUNT(*)
    FROM EXTRA_OR_CHANGED_ROWS

    UNION ALL
    SELECT '11 Preview Open findings', COUNT(DISTINCT FINDING_KEY)
    FROM PREVIEW_OUTPUT
    WHERE UPPER(TRIM(COALESCE(OVERALL_STATUS, ''))) = 'OPEN'

    UNION ALL
    SELECT '12 Preview Closed findings', COUNT(DISTINCT FINDING_KEY)
    FROM PREVIEW_OUTPUT
    WHERE UPPER(TRIM(COALESCE(OVERALL_STATUS, ''))) = 'CLOSED'

    UNION ALL
    SELECT '13 Preview Draft findings', COUNT(DISTINCT FINDING_KEY)
    FROM PREVIEW_OUTPUT
    WHERE UPPER(TRIM(COALESCE(OVERALL_STATUS, ''))) = 'DRAFT'

    UNION ALL
    SELECT '14 Preview other or blank status', COUNT(DISTINCT FINDING_KEY)
    FROM PREVIEW_OUTPUT
    WHERE UPPER(TRIM(COALESCE(OVERALL_STATUS, '')))
          NOT IN ('OPEN', 'CLOSED', 'DRAFT')
)

SELECT
    L.CUTOFF AS CURRENT_SOURCE_CUTOFF,
    C.CHECK_NAME,
    C.RESULT
FROM CHECKS C
CROSS JOIN LATEST L
ORDER BY C.CHECK_NAME;
