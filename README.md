-- READ ONLY: identifies column-level differences.
-- Replace PASTE_PREVIEW_QUERY_ID only.

WITH PREVIEW_OUTPUT AS (
    SELECT *
    FROM TABLE(RESULT_SCAN('PASTE_PREVIEW_QUERY_ID'))
),

ROW_OBJECTS AS (
    SELECT
        'EXISTING' AS QA_SIDE,
        V.REMEDIATION_PLAN_KEY AS PLAN_KEY,
        OBJECT_CONSTRUCT_KEEP_NULL(V.*) AS ROW_DATA
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.C2S_ENTERPRISE_REMEDIATION_PLANS_DATA V
    WHERE EXISTS (
        SELECT 1
        FROM PREVIEW_OUTPUT P
        WHERE P.REMEDIATION_PLAN_KEY = V.REMEDIATION_PLAN_KEY
    )

    UNION ALL

    SELECT
        'PREVIEW' AS QA_SIDE,
        P.REMEDIATION_PLAN_KEY AS PLAN_KEY,
        OBJECT_CONSTRUCT_KEEP_NULL(P.*) AS ROW_DATA
    FROM PREVIEW_OUTPUT P
),

VALUE_COUNTS AS (
    SELECT
        R.PLAN_KEY,
        F.KEY::VARCHAR AS COLUMN_NAME,
        F.VALUE AS FIELD_VALUE,
        SUM(IFF(R.QA_SIDE = 'EXISTING', 1, 0)) AS EXISTING_COPIES,
        SUM(IFF(R.QA_SIDE = 'PREVIEW', 1, 0)) AS PREVIEW_COPIES
    FROM ROW_OBJECTS R,
         LATERAL FLATTEN(INPUT => R.ROW_DATA) F
    GROUP BY
        R.PLAN_KEY,
        F.KEY::VARCHAR,
        F.VALUE
),

DIFFERENCES AS (
    SELECT *
    FROM VALUE_COUNTS
    WHERE EXISTING_COPIES <> PREVIEW_COPIES
),

COLUMN_SUMMARY AS (
    SELECT
        COLUMN_NAME,
        COUNT(DISTINCT PLAN_KEY) AS AFFECTED_PLANS,
        MIN(PLAN_KEY) AS EXAMPLE_PLAN_KEY
    FROM DIFFERENCES
    GROUP BY COLUMN_NAME
)

SELECT
    S.COLUMN_NAME,
    S.AFFECTED_PLANS,
    S.EXAMPLE_PLAN_KEY,
    D.FIELD_VALUE,
    D.EXISTING_COPIES,
    D.PREVIEW_COPIES
FROM COLUMN_SUMMARY S
JOIN DIFFERENCES D
  ON D.COLUMN_NAME = S.COLUMN_NAME
 AND D.PLAN_KEY = S.EXAMPLE_PLAN_KEY
ORDER BY
    S.COLUMN_NAME,
    D.EXISTING_COPIES DESC,
    D.PREVIEW_COPIES DESC;
