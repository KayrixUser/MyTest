CREATE OR REPLACE VIEW
    RTX_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.C2S_SOX_FINANCE_TREND_HISTORY
COPY GRANTS
AS
WITH BASE AS (
    SELECT
        CAPTURE_DATE,
        CAPTURED_AT_UTC,
        BUSINESS_UNIT_NAME,
        COMMITTED_CLOSURE_DATE,
        CASE
            WHEN LOWER(TRIM(OVERALL_STATUS)) = 'in process'
            THEN FINDING_ID
        END AS IN_PROCESS_FINDING_ID,
        DATEDIFF('day', COMMITTED_CLOSURE_DATE, CAPTURE_DATE)
            AS DAYS_PAST_DUE
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.C2S_SOX_FINANCE_TREND_HISTORY
),
RULES AS (
    SELECT
        *,
        DAYS_PAST_DUE <= 30 AS MET_RULE,
        (
            (
                YEAR(COMMITTED_CLOSURE_DATE) = YEAR(CAPTURE_DATE)
                AND DAYS_PAST_DUE > 30
            )
            OR YEAR(COMMITTED_CLOSURE_DATE) <> YEAR(CAPTURE_DATE)
        ) AS RISK_RULE
    FROM BASE
)
SELECT
    CAPTURE_DATE,
    CAPTURED_AT_UTC,
    BUSINESS_UNIT_NAME,

    COUNT(DISTINCT IN_PROCESS_FINDING_ID)
        AS TOTAL_IN_PROCESS_FINDINGS,

    COUNT(DISTINCT CASE
        WHEN MET_RULE THEN IN_PROCESS_FINDING_ID
    END) AS TOTAL_FINDINGS_TARGET_MET,

    COUNT(DISTINCT CASE
        WHEN RISK_RULE THEN IN_PROCESS_FINDING_ID
    END) AS TOTAL_FINDINGS_TARGET_AT_RISK,

    COUNT(DISTINCT CASE
        WHEN COMMITTED_CLOSURE_DATE IS NULL
        THEN IN_PROCESS_FINDING_ID
    END) AS MISSING_CLOSURE_DATE,

    COUNT(DISTINCT CASE
        WHEN MET_RULE THEN IN_PROCESS_FINDING_ID
    END)
    + COUNT(DISTINCT CASE
        WHEN RISK_RULE THEN IN_PROCESS_FINDING_ID
    END)
    - COUNT(DISTINCT CASE
        WHEN MET_RULE OR RISK_RULE THEN IN_PROCESS_FINDING_ID
    END) AS OVERLAPPING_FINDINGS

FROM RULES
GROUP BY
    CAPTURE_DATE,
    CAPTURED_AT_UTC,
    BUSINESS_UNIT_NAME;

;WITH SourceTLS AS (
    SELECT hostname, fqdn, Total_TLS_Vulnerabilities, TLS_Encryption
    FROM dbo.view_c2sreport_tbl_TLSVerdant
),
LoadedTLS AS (
    SELECT hostname, fqdn, Total_TLS_Vulnerabilities, TLS_Encryption
    FROM dbo.dimTLSVerdant
),
FactLinks AS (
    SELECT DISTINCT HWID, TLSID
    FROM dbo.factCyberEvents
),
ExpectedLinks AS (
    SELECT DISTINCT H.HWID, T.TLSID
    FROM (SELECT DISTINCT HWID FROM FactLinks) AS F
    JOIN dbo.dimHardware AS H ON H.HWID = F.HWID
    LEFT JOIN dbo.dimTLSVerdant AS T
        ON dbo.ParseStringValue(T.hostname) = H.Archer_Asset_Name
)
SELECT
    (SELECT COUNT_BIG(*) FROM SourceTLS) AS SourceRows,
    (SELECT COUNT_BIG(*) FROM LoadedTLS) AS LoadedRows,
    (SELECT COUNT_BIG(*) FROM (
        SELECT * FROM SourceTLS EXCEPT SELECT * FROM LoadedTLS
    ) AS X) AS SourceOnly,
    (SELECT COUNT_BIG(*) FROM (
        SELECT * FROM LoadedTLS EXCEPT SELECT * FROM SourceTLS
    ) AS X) AS LoadedOnly,
    (SELECT COUNT_BIG(*) FROM (
        SELECT * FROM ExpectedLinks EXCEPT SELECT * FROM FactLinks
    ) AS X) AS MissingFactLinks,
    (SELECT COUNT_BIG(*) FROM (
        SELECT * FROM FactLinks EXCEPT SELECT * FROM ExpectedLinks
    ) AS X) AS UnexpectedFactLinks;
