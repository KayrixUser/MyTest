-- READ ONLY. Run after the Findings load has completed.

WITH LATEST AS (
    SELECT
        MAX(ARCHER_CONTENT_FINDINGS_CONFIRMED_IN_ARCHER) AS CUTOFF
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_FINDINGS
),

OPEN_FINDINGS AS (
    SELECT
        F.ARCHER_CONTENT_FINDINGS_CONTENT_ID AS FINDING_KEY,
        TO_VARCHAR(F.POLICY_LEVEL_3) AS POLICY_IDS_RAW
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_FINDINGS F
    CROSS JOIN LATEST L
    WHERE F.ARCHER_CONTENT_FINDINGS_CONFIRMED_IN_ARCHER = L.CUTOFF
      AND F.OVERALL_STATUS = 'Open'
),

TOKENS AS (
    SELECT
        F.FINDING_KEY,
        TRIM(S.VALUE) AS POLICY_ID_RAW
    FROM OPEN_FINDINGS F,
         LATERAL SPLIT_TO_TABLE(
             COALESCE(F.POLICY_IDS_RAW, ''), ';'
         ) S
    WHERE NULLIF(TRIM(S.VALUE), '') IS NOT NULL
),

PARSED AS (
    SELECT
        FINDING_KEY,
        POLICY_ID_RAW,
        CASE
            WHEN REGEXP_LIKE(POLICY_ID_RAW, '^[0-9]+$')
            THEN TRY_TO_NUMBER(POLICY_ID_RAW)
        END AS POLICY_KEY
    FROM TOKENS
),

POLICY_PAIRS AS (
    SELECT DISTINCT FINDING_KEY, POLICY_KEY
    FROM PARSED
    WHERE POLICY_KEY IS NOT NULL
),

POLICY_LOOKUP AS (
    SELECT
        ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3_CONTENT_ID AS POLICY_KEY,
        NULLIF(TRIM(POLICY_LEVEL_3_NAME), '') AS POLICY_REFERENCE,
        NULLIF(TRIM(POLICY_NAME), '') AS POLICY_TITLE
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3
),

RESOLVED AS (
    SELECT
        R.FINDING_KEY,
        R.POLICY_KEY,
        P.POLICY_KEY AS MATCHED_POLICY_KEY,
        P.POLICY_REFERENCE,
        P.POLICY_TITLE
    FROM POLICY_PAIRS R
    LEFT JOIN POLICY_LOOKUP P
      ON P.POLICY_KEY = R.POLICY_KEY
),

DUPLICATE_LOOKUP_KEYS AS (
    SELECT P.POLICY_KEY
    FROM POLICY_LOOKUP P
    WHERE P.POLICY_KEY IN (
        SELECT POLICY_KEY FROM POLICY_PAIRS
    )
    GROUP BY P.POLICY_KEY
    HAVING COUNT(*) > 1
),

CHECKS AS (
    SELECT '01 Latest Open findings' AS CHECK_NAME,
           COUNT(DISTINCT FINDING_KEY) AS RESULT
    FROM OPEN_FINDINGS

    UNION ALL
    SELECT '02 Open findings with blank direct policy field',
           COUNT(DISTINCT FINDING_KEY)
    FROM OPEN_FINDINGS
    WHERE NULLIF(TRIM(POLICY_IDS_RAW), '') IS NULL

    UNION ALL
    SELECT '03 Invalid policy ID tokens', COUNT(*)
    FROM PARSED
    WHERE POLICY_KEY IS NULL

    UNION ALL
    SELECT '04 Referenced policy keys missing from lookup',
           COUNT(DISTINCT POLICY_KEY)
    FROM RESOLVED
    WHERE MATCHED_POLICY_KEY IS NULL

    UNION ALL
    SELECT '05 Referenced policy keys duplicated in lookup', COUNT(*)
    FROM DUPLICATE_LOOKUP_KEYS

    UNION ALL
    SELECT '06 Matched pairs with blank policy reference', COUNT(*)
    FROM RESOLVED
    WHERE MATCHED_POLICY_KEY IS NOT NULL
      AND POLICY_REFERENCE IS NULL

    UNION ALL
    SELECT '07 Matched pairs with blank policy title', COUNT(*)
    FROM RESOLVED
    WHERE MATCHED_POLICY_KEY IS NOT NULL
      AND POLICY_TITLE IS NULL

    UNION ALL
    SELECT '08 Distinct source Finding-policy pairs', COUNT(*)
    FROM POLICY_PAIRS

    UNION ALL
    SELECT '09 Resolved Finding-policy pairs',
           COUNT(DISTINCT FINDING_KEY, POLICY_KEY)
    FROM RESOLVED
    WHERE MATCHED_POLICY_KEY IS NOT NULL
      AND POLICY_REFERENCE IS NOT NULL

    UNION ALL
    SELECT '10 Distinct resolved policy keys',
           COUNT(DISTINCT POLICY_KEY)
    FROM RESOLVED
    WHERE MATCHED_POLICY_KEY IS NOT NULL
      AND POLICY_REFERENCE IS NOT NULL

    UNION ALL
    SELECT '11 Distinct resolved policy references',
           COUNT(DISTINCT POLICY_REFERENCE)
    FROM RESOLVED
    WHERE MATCHED_POLICY_KEY IS NOT NULL
      AND POLICY_REFERENCE IS NOT NULL

    UNION ALL
    SELECT '12 Open findings with resolved policies',
           COUNT(DISTINCT FINDING_KEY)
    FROM RESOLVED
    WHERE MATCHED_POLICY_KEY IS NOT NULL
      AND POLICY_REFERENCE IS NOT NULL
)

SELECT
    L.CUTOFF AS CURRENT_SOURCE_CUTOFF,
    C.CHECK_NAME,
    C.RESULT
FROM CHECKS C
CROSS JOIN LATEST L
ORDER BY C.CHECK_NAME;
