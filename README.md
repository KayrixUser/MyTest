SELECT
    COUNT(*) OVER () AS SOURCE_ROWS,
    p.Content_ID AS CONTENT_ID, p.POAM_ID, p.Name AS NAME, p.Overall_Status AS OVERALL_STATUS,
    p.Authorization_Package AS PACKAGE_NAME,
    p.POAM_Owner AS OWNER_RAW, p.POAM_Owner_Name AS OWNER_NAME, p.POAM_Owner_UID AS OWNER_UID,
    p.Related_Allocated_Controls AS RELATED_CONTROLS,
    p.Last_Updated AS LAST_UPDATED,
    CONVERT(varchar(23), p.ConfirmedInArcher, 126) AS CONFIRMED_IN_ARCHER,
    CONVERT(varchar(23), p.First_Published, 126) AS FIRST_PUBLISHED,
    p.hRTN_First_Published AS HISTORICAL_FIRST_PUBLISHED,
    CONVERT(varchar(23), p.Milestone_Estimated_Completion_Date, 126) AS MILESTONE_DATE,
    CONVERT(varchar(23), p.POAM_Target_Completion_Date, 126) AS TARGET_DATE,
    d.Published AS EFFECTIVE_PUBLISH_DATE,
    DATEDIFF(day, d.Published, CAST('20261124' AS date)) AS DAYS_OPEN
FROM ACP_Connection.dbo.ArcherMetrics_POAMs_PROD_All AS p
CROSS APPLY (VALUES (
    CASE WHEN NULLIF(TRIM(p.hRTN_First_Published), '') IS NULL
         THEN CAST(p.First_Published AS date)
         ELSE TRY_CONVERT(date, p.hRTN_First_Published) END
)) AS d(Published)
ORDER BY p.Content_ID;


WITH poams AS (
    SELECT *,
        CASE WHEN NULLIF(TRIM(TO_VARCHAR(HRTN_FIRST_PUBLISHED)), '') IS NULL
             THEN TO_DATE(ARCHER_CONTENT_POAMS_FIRST_PUBLISHED)
             ELSE TO_DATE(TRY_TO_TIMESTAMP_NTZ(TO_VARCHAR(HRTN_FIRST_PUBLISHED)))
        END AS EFFECTIVE_PUBLISH_DATE
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.C2S_POAM
)
SELECT
    COUNT(*) OVER () AS SOURCE_ROWS,
    TO_VARCHAR(ARCHER_CONTENT_POAMS_CONTENT_ID) AS CONTENT_ID, POAM_ID, NAME, OVERALL_STATUS,
    AUTHORIZATION_PACKAGE_NAME AS PACKAGE_NAME,
    POAM_OWNER AS OWNER_RAW, POAM_OWNER_NAME_ONLY AS OWNER_NAME, POAM_OWNER_ID_ONLY AS OWNER_UID,
    RELATED_ALLOCATED_CONTROLS AS RELATED_CONTROLS,
    ARCHER_CONTENT_POAMS_LAST_UPDATED AS LAST_UPDATED,
    ARCHER_CONTENT_POAMS_CONFIRMED_IN_ARCHER AS CONFIRMED_IN_ARCHER,
    ARCHER_CONTENT_POAMS_FIRST_PUBLISHED AS FIRST_PUBLISHED,
    HRTN_FIRST_PUBLISHED AS HISTORICAL_FIRST_PUBLISHED,
    MILESTONE_ESTIMATED_COMPLETION_DATE AS MILESTONE_DATE,
    POAM_TARGET_COMPLETION_DATE AS TARGET_DATE,
    EFFECTIVE_PUBLISH_DATE,
    DATEDIFF(day, EFFECTIVE_PUBLISH_DATE, DATE '2026-11-24') AS DAYS_OPEN
FROM poams
ORDER BY CONTENT_ID;
