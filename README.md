EXPECTED_ROWS AS (
    SELECT
        E.* REPLACE (
            ARRAY_SORT(SPLIT(
                E.REMEDIATION_PLAN_OWNER_ID_NAME, ';'
            )) AS REMEDIATION_PLAN_OWNER_ID_NAME,

            ARRAY_SORT(SPLIT(
                E.REMEDIATION_PLAN_OWNER_ID_ONLY, ';'
            )) AS REMEDIATION_PLAN_OWNER_ID_ONLY,

            ARRAY_SORT(SPLIT(
                E.REMEDIATION_PLAN_OWNER_NAME_ONLY, ';'
            )) AS REMEDIATION_PLAN_OWNER_NAME_ONLY
        ),
        COUNT(*) AS QA_ROW_COPIES
    FROM EXPECTED_OUTPUT E
    GROUP BY ALL
),

PREVIEW_ROWS AS (
    SELECT
        P.* REPLACE (
            ARRAY_SORT(SPLIT(
                P.REMEDIATION_PLAN_OWNER_ID_NAME, ';'
            )) AS REMEDIATION_PLAN_OWNER_ID_NAME,

            ARRAY_SORT(SPLIT(
                P.REMEDIATION_PLAN_OWNER_ID_ONLY, ';'
            )) AS REMEDIATION_PLAN_OWNER_ID_ONLY,

            ARRAY_SORT(SPLIT(
                P.REMEDIATION_PLAN_OWNER_NAME_ONLY, ';'
            )) AS REMEDIATION_PLAN_OWNER_NAME_ONLY
        ),
        COUNT(*) AS QA_ROW_COPIES
    FROM PREVIEW_OUTPUT P
    GROUP BY ALL
),
