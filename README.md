/* READ ONLY — run this entire WITH ... SELECT statement together.
   No view/table/Power BI changes.
   Scope: currently Open records in shared ENTERPRISE_FINDINGS.
   Route: CONTROL_STANDARDS -> Control Standard Content ID
          -> Control Standard.POLICY_LEVEL_3_NAME (recorded ID list)
          -> Policy Level 3 Content ID -> policy reference code.
   Null/missing links are retained for validation, not silently discarded.
   Blank descriptive titles are reported separately from missing policy codes.
*/
WITH OPEN_FINDINGS AS (
    SELECT
        ARCHER_CONTENT_FINDINGS_CONTENT_ID AS FINDING_KEY,
        CONTROL_STANDARDS
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.ENTERPRISE_FINDINGS
    WHERE OVERALL_STATUS = 'Open'
),
CONTROL_REFERENCES AS (
    SELECT DISTINCT
        F.FINDING_KEY,
        NULLIF(TRIM(S.VALUE), '') AS CONTROL_ID_RAW
    FROM OPEN_FINDINGS F,
         LATERAL SPLIT_TO_TABLE(
             REPLACE(COALESCE(TO_VARCHAR(F.CONTROL_STANDARDS), ''), ',', ';'),
             ';'
         ) S
),
CONTROL_LINKS AS (
    SELECT
        R.FINDING_KEY,
        R.CONTROL_ID_RAW,
        C.ARCHER_CONTENT_CONTROL_STANDARDS_CONTENT_ID AS MATCHED_CONTROL_KEY,
        C.POLICY_LEVEL_3_NAME AS POLICY_IDS_RAW
    FROM CONTROL_REFERENCES R
    LEFT JOIN RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_CONTROL_STANDARDS C
        ON TRY_TO_NUMBER(R.CONTROL_ID_RAW)
           = C.ARCHER_CONTENT_CONTROL_STANDARDS_CONTENT_ID
),
POLICY_REFERENCES AS (
    SELECT
        C.FINDING_KEY,
        C.CONTROL_ID_RAW,
        C.MATCHED_CONTROL_KEY,
        NULLIF(TRIM(S.VALUE), '') AS POLICY_ID_RAW
    FROM CONTROL_LINKS C,
         LATERAL SPLIT_TO_TABLE(
             REPLACE(COALESCE(TO_VARCHAR(C.POLICY_IDS_RAW), ''), ',', ';'),
             ';'
         ) S
),
RESOLVED AS (
    SELECT
        R.*,
        P.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3_CONTENT_ID AS MATCHED_POLICY_KEY,
        NULLIF(TRIM(P.POLICY_LEVEL_3_NAME), '') AS POLICY_REFERENCE,
        NULLIF(TRIM(P.POLICY_NAME), '') AS POLICY_TITLE
    FROM POLICY_REFERENCES R
    LEFT JOIN RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3 P
        ON TRY_TO_NUMBER(R.POLICY_ID_RAW)
           = P.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3_CONTENT_ID
),
VALID_PAIRS AS (
    /* Two standards from one Finding that lead to the same policy count once. */
    SELECT DISTINCT FINDING_KEY, MATCHED_POLICY_KEY
    FROM RESOLVED
    WHERE FINDING_KEY IS NOT NULL
      AND MATCHED_CONTROL_KEY IS NOT NULL
      AND MATCHED_POLICY_KEY IS NOT NULL
      AND POLICY_REFERENCE IS NOT NULL
),
DUPLICATE_CONTROL_KEYS AS (
    SELECT C.ARCHER_CONTENT_CONTROL_STANDARDS_CONTENT_ID AS CONTROL_KEY
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_CONTROL_STANDARDS C
    WHERE C.ARCHER_CONTENT_CONTROL_STANDARDS_CONTENT_ID IN (
        SELECT MATCHED_CONTROL_KEY FROM CONTROL_LINKS
    )
    GROUP BY C.ARCHER_CONTENT_CONTROL_STANDARDS_CONTENT_ID
    HAVING COUNT(*) > 1
),
DUPLICATE_POLICY_KEYS AS (
    SELECT P.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3_CONTENT_ID AS POLICY_KEY
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3 P
    WHERE P.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3_CONTENT_ID IN (
        SELECT MATCHED_POLICY_KEY FROM RESOLVED
    )
    GROUP BY P.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3_CONTENT_ID
    HAVING COUNT(*) > 1
)
SELECT '01 Shared-view Open rows' AS CHECK_NAME,
       TO_VARCHAR(COUNT(*)) AS RESULT
FROM OPEN_FINDINGS
UNION ALL
SELECT '02 Distinct Open Finding keys', TO_VARCHAR(COUNT(DISTINCT FINDING_KEY))
FROM OPEN_FINDINGS
UNION ALL
SELECT '03 Open rows with null Finding key', TO_VARCHAR(COUNT(*))
FROM OPEN_FINDINGS WHERE FINDING_KEY IS NULL
UNION ALL
SELECT '04 Open Findings with no nonblank Control Standard reference',
       TO_VARCHAR(COUNT(DISTINCT FINDING_KEY))
FROM OPEN_FINDINGS F
WHERE NOT EXISTS (
    SELECT 1 FROM CONTROL_REFERENCES R
    WHERE R.FINDING_KEY = F.FINDING_KEY AND R.CONTROL_ID_RAW IS NOT NULL
)
UNION ALL
SELECT '05 Unresolved Control Standard reference values',
       TO_VARCHAR(COUNT(DISTINCT CONTROL_ID_RAW))
FROM CONTROL_LINKS
WHERE CONTROL_ID_RAW IS NOT NULL AND MATCHED_CONTROL_KEY IS NULL
UNION ALL
SELECT '06 Matched Control Standards with no nonblank Policy Level 3 reference',
       TO_VARCHAR(COUNT(DISTINCT C.MATCHED_CONTROL_KEY))
FROM CONTROL_LINKS C
WHERE C.MATCHED_CONTROL_KEY IS NOT NULL
  AND NOT EXISTS (
      SELECT 1 FROM POLICY_REFERENCES R
      WHERE R.MATCHED_CONTROL_KEY = C.MATCHED_CONTROL_KEY
        AND R.POLICY_ID_RAW IS NOT NULL
  )
UNION ALL
SELECT '07 Unresolved Policy Level 3 reference values',
       TO_VARCHAR(COUNT(DISTINCT POLICY_ID_RAW))
FROM RESOLVED
WHERE POLICY_ID_RAW IS NOT NULL AND MATCHED_POLICY_KEY IS NULL
UNION ALL
SELECT '08 Matched policies missing display codes',
       TO_VARCHAR(COUNT(DISTINCT MATCHED_POLICY_KEY))
FROM RESOLVED
WHERE MATCHED_POLICY_KEY IS NOT NULL AND POLICY_REFERENCE IS NULL
UNION ALL
SELECT '09 Referenced duplicate Control Standard master keys', TO_VARCHAR(COUNT(*))
FROM DUPLICATE_CONTROL_KEYS
UNION ALL
SELECT '10 Referenced duplicate Policy Level 3 master keys', TO_VARCHAR(COUNT(*))
FROM DUPLICATE_POLICY_KEYS
UNION ALL
SELECT '11 Open Findings represented through valid policy-code links',
       TO_VARCHAR(COUNT(DISTINCT FINDING_KEY))
FROM VALID_PAIRS
UNION ALL
SELECT '12 Open Findings without a valid policy-code link',
       TO_VARCHAR(COUNT(DISTINCT F.FINDING_KEY))
FROM OPEN_FINDINGS F
WHERE NOT EXISTS (
    SELECT 1 FROM VALID_PAIRS V WHERE V.FINDING_KEY = F.FINDING_KEY
)
UNION ALL
SELECT '13 Distinct Open Finding-Policy pairs', TO_VARCHAR(COUNT(*))
FROM VALID_PAIRS
UNION ALL
SELECT '14 Matched policies with codes but missing descriptive titles',
       TO_VARCHAR(COUNT(DISTINCT MATCHED_POLICY_KEY))
FROM RESOLVED
WHERE MATCHED_POLICY_KEY IS NOT NULL
  AND POLICY_REFERENCE IS NOT NULL AND POLICY_TITLE IS NULL
UNION ALL
SELECT '15 Sample policy 7250297 - resolved master code',
       COALESCE(
           NULLIF(
               LISTAGG(DISTINCT NULLIF(TRIM(POLICY_LEVEL_3_NAME), ''), '; ')
                   WITHIN GROUP (ORDER BY NULLIF(TRIM(POLICY_LEVEL_3_NAME), '')),
               ''
           ),
           'NO POPULATED CODE FOUND'
       )
FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3
WHERE ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3_CONTENT_ID = 7250297
UNION ALL
SELECT '16 Sample policy 7252752 - resolved master code',
       COALESCE(
           NULLIF(
               LISTAGG(DISTINCT NULLIF(TRIM(POLICY_LEVEL_3_NAME), ''), '; ')
                   WITHIN GROUP (ORDER BY NULLIF(TRIM(POLICY_LEVEL_3_NAME), '')),
               ''
           ),
           'NO POPULATED CODE FOUND'
       )
FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3
WHERE ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3_CONTENT_ID = 7252752
ORDER BY CHECK_NAME;
