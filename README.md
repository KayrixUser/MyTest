WITH REPORTING_ROWS_TO_TEST AS (
WITH BASE AS (
    SELECT
        P.ARCHER_CONTENT_REMEDIATION_PLANS_CONTENT_ID AS REMEDIATION_PLAN_KEY,
        P.REMEDIATION_PLAN_ID,
        P.REMEDIATION_PLAN_NAME,
        P.REMEDIATION_PLAN_DESCRIPTION,
        P.REMEDIATION_TYPE,
        P.STATUS,
        P.CURRENT_STATUS,
        P.CURRENT_AS_OF_DATE,
        R.RECORD_STATUS,
        R.REMEDIATION_COMPLETION_STATUS,
        P.ESTIMATED_COMPLETION_DATE,
        P.ORIGINAL_ESTIMATED_COMPLETION_DATE,
        R.ESTIMATED_COMPLETION_DATE_FOR_APPROVED_REMEDIATION_PLANS
            AS APPROVED_PLAN_ESTIMATED_COMPLETION_DATE,
        P.ACTUAL_COMPLETION_DATE,
        P.ESTIMATED_START_DATE,
        P.ACTUAL_START_DATE,
        P.SUBMISSION_DUE_DATE,
        P.REVIEW_DUE_DATE,
        P.REMEDIATION_PLAN_OWNER,
        P.REMEDIATION_PLAN_OWNER_ID_ONLY,
        P.REMEDIATION_PLAN_OWNER_NAME_ONLY,
        P.REMEDIATION_PLAN_OWNER_ID_NAME,
        P.BU_ASSIGNEE_ID_ONLY,
        P.BU_ASSIGNEE_NAME_ONLY,
        R.BUSINESS_UNIT AS RP_BUSINESS_UNIT_RAW,
        R.SUBORGANIZATION AS RP_SBU_RAW,
        R.ENTITY AS RP_ENTITY_RAW,
        P.RELATED_FINDINGS_CONTENT_IDS,
        P.RELATED_FINDING_IDS,
        P.RELATED_FINDING_BUSINESS_UNIT,
        P.PRIORITY_RATING_SCALE,
        P.PRIORITY_RATING,
        P.OVERDUE_REASON,
        P.ARCHER_CONTENT_REMEDIATION_PLANS_FIRST_PUBLISHED AS ARCHER_FIRST_PUBLISHED_AT,
        P.ARCHER_CONTENT_REMEDIATION_PLANS_LAST_UPDATED AS ARCHER_LAST_UPDATED_AT,
        R.ARCHER_CONTENT_REMEDIATION_PLANS_CONTENT_ID IS NOT NULL AS RP_SOURCE_MATCHED
    FROM ABC_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.ENTERPRISE_REMEDIATION_PLANS P
    LEFT JOIN ABC_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_REMEDIATION_PLANS R
        ON P.ARCHER_CONTENT_REMEDIATION_PLANS_CONTENT_ID
           = R.ARCHER_CONTENT_REMEDIATION_PLANS_CONTENT_ID
),
-- Each input row describes ONE kind of plan-level organizational assignment.
ORG_INPUT AS (
    SELECT REMEDIATION_PLAN_KEY, 'BU' AS ORG_TYPE,
           RP_BUSINESS_UNIT_RAW AS RAW_IDS FROM BASE
    UNION ALL
    SELECT REMEDIATION_PLAN_KEY, 'SBU', RP_SBU_RAW FROM BASE
    UNION ALL
    SELECT REMEDIATION_PLAN_KEY, 'ENTITY', RP_ENTITY_RAW FROM BASE
),
TOKEN_VALUES AS (
    SELECT
        I.REMEDIATION_PLAN_KEY,
        I.ORG_TYPE,
        TRIM(S.VALUE) AS TOKEN_RAW,
        CASE WHEN REGEXP_LIKE(TRIM(S.VALUE), '[0-9]+')
             THEN TRY_TO_NUMBER(TRIM(S.VALUE)) END AS ORG_KEY
    FROM ORG_INPUT I,
         LATERAL SPLIT_TO_TABLE(
             REPLACE(COALESCE(I.RAW_IDS, ''), ';', ','), ','
         ) S
    WHERE NULLIF(TRIM(S.VALUE), '') IS NOT NULL
),
-- Deduplicate repeated references within each plan and organization type.
-- Malformed references are retained for explicit unresolved labels.
ORG_TOKENS AS (
    SELECT DISTINCT
        REMEDIATION_PLAN_KEY, ORG_TYPE, ORG_KEY,
        COALESCE(TO_VARCHAR(ORG_KEY), TOKEN_RAW) AS TOKEN_ID
    FROM TOKEN_VALUES
),
MASTER_INPUT AS (
    SELECT 'BU' AS ORG_TYPE,
           ARCHER_CONTENT_BUSINESS_CONTENT_ID AS ORG_KEY,
           NULLIF(TRIM(BUSINESS), '') AS ORG_NAME
    FROM ABC_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_BUSINESS
    UNION ALL
    SELECT 'SBU', ARCHER_CONTENT_SUBORGANIZATION_CONTENT_ID,
           NULLIF(TRIM(SUBORGANIZATION), '')
    FROM ABC_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_SUBORGANIZATION
    UNION ALL

           NULLIF(TRIM(ENTITY), '')
    FROM ABC_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_ENTITY
),
-- Do not arbitrarily choose one label for an ID with conflicting names.
MASTER_BY_KEY AS (
    SELECT
        ORG_TYPE, ORG_KEY,
        COUNT(DISTINCT ORG_NAME) AS NAME_COUNT,
        CASE WHEN COUNT(DISTINCT ORG_NAME) = 1
             THEN MAX(ORG_NAME) END AS ORG_NAME
    FROM MASTER_INPUT
    WHERE ORG_KEY IS NOT NULL
    GROUP BY ORG_TYPE, ORG_KEY
),
-- A slicer must distinguish different IDs sharing the same readable name.
MASTER AS (
    SELECT M.*,
           COUNT(*) OVER (
               PARTITION BY ORG_TYPE, UPPER(ORG_NAME)
           ) AS KEYS_WITH_SAME_NAME
    FROM MASTER_BY_KEY M
),
ORG_RESOLVED AS (
    SELECT
        T.REMEDIATION_PLAN_KEY,
        T.ORG_TYPE,
        T.ORG_KEY,
        T.TOKEN_ID,
        CASE
            WHEN T.ORG_KEY IS NULL THEN 'INVALID_REFERENCE'
            WHEN M.ORG_KEY IS NULL THEN 'UNMATCHED_REFERENCE'
            WHEN M.NAME_COUNT > 1 THEN 'CONFLICTING_NAMES'
            WHEN M.NAME_COUNT = 0 THEN 'MISSING_NAME'
            ELSE 'RESOLVED'
        END AS RESOLUTION_STATUS,
        CASE
            WHEN T.ORG_KEY IS NULL OR M.ORG_KEY IS NULL
                THEN 'Unresolved ' || T.ORG_TYPE || ' [' || T.TOKEN_ID || ']'
            WHEN M.NAME_COUNT > 1
                THEN 'Conflicting ' || T.ORG_TYPE || ' names [' || T.TOKEN_ID || ']'
            WHEN M.NAME_COUNT = 0
                THEN 'Missing ' || T.ORG_TYPE || ' name [' || T.TOKEN_ID || ']'
            WHEN M.KEYS_WITH_SAME_NAME > 1
                THEN M.ORG_NAME || ' [' || T.TOKEN_ID || ']'
            ELSE M.ORG_NAME
        END AS ORG_LABEL
    FROM ORG_TOKENS T
    LEFT JOIN MASTER M
        ON M.ORG_TYPE = T.ORG_TYPE
       AND M.ORG_KEY = T.ORG_KEY
),
-- Constant full-assignment lists for the one-plan-per-line detail table.
-- These are DISPLAY fields, not the fields for individual-organization slicers.
ORG_LISTS AS (
    SELECT REMEDIATION_PLAN_KEY, ORG_TYPE,
           LISTAGG(DISTINCT ORG_LABEL, '; ')
               WITHIN GROUP (ORDER BY ORG_LABEL) AS ORG_NAME_LIST
    FROM ORG_RESOLVED
    GROUP BY REMEDIATION_PLAN_KEY, ORG_TYPE
)
SELECT
    B.*,
    BU.ORG_KEY AS RP_BUSINESS_KEY,
    SBU.ORG_KEY AS RP_SBU_KEY,
    ENT.ORG_KEY AS RP_ENTITY_KEY,
    COALESCE(BU.ORG_LABEL, 'Not recorded') AS RP_BUSINESS_NAME,
    COALESCE(SBU.ORG_LABEL, 'Not recorded') AS RP_SBU_NAME,
    COALESCE(ENT.ORG_LABEL, 'Not recorded') AS RP_ENTITY_NAME,
    COALESCE(BUL.ORG_NAME_LIST, 'Not recorded') AS RP_BUSINESS_NAMES_ALL,
    COALESCE(SBUL.ORG_NAME_LIST, 'Not recorded') AS RP_SBU_NAMES_ALL,
    COALESCE(ENTL.ORG_NAME_LIST, 'Not recorded') AS RP_ENTITY_NAMES_ALL,
    COALESCE(BU.RESOLUTION_STATUS, 'NOT_RECORDED') AS RP_BUSINESS_LOOKUP_STATUS,
    COALESCE(SBU.RESOLUTION_STATUS, 'NOT_RECORDED') AS RP_SBU_LOOKUP_STATUS,
    COALESCE(ENT.RESOLUTION_STATUS, 'NOT_RECORDED') AS RP_ENTITY_LOOKUP_STATUS
FROM BASE B
-- These independent same-plan joins expand membership selections.
-- They DO NOT establish a Business -> SBU -> Entity master hierarchy.
LEFT JOIN ORG_RESOLVED BU
    ON BU.REMEDIATION_PLAN_KEY = B.REMEDIATION_PLAN_KEY AND BU.ORG_TYPE = 'BU'
LEFT JOIN ORG_RESOLVED SBU
    ON SBU.REMEDIATION_PLAN_KEY = B.REMEDIATION_PLAN_KEY AND SBU.ORG_TYPE = 'SBU'
LEFT JOIN ORG_RESOLVED ENT
    ON ENT.REMEDIATION_PLAN_KEY = B.REMEDIATION_PLAN_KEY AND ENT.ORG_TYPE = 'ENTITY'
LEFT JOIN ORG_LISTS BUL
    ON BUL.REMEDIATION_PLAN_KEY = B.REMEDIATION_PLAN_KEY AND BUL.ORG_TYPE = 'BU'
LEFT JOIN ORG_LISTS SBUL
    ON SBUL.REMEDIATION_PLAN_KEY = B.REMEDIATION_PLAN_KEY AND SBUL.ORG_TYPE = 'SBU'
LEFT JOIN ORG_LISTS ENTL
    ON ENTL.REMEDIATION_PLAN_KEY = B.REMEDIATION_PLAN_KEY AND ENTL.ORG_TYPE = 'ENTITY'
)
SELECT
    (SELECT COUNT(DISTINCT ARCHER_CONTENT_REMEDIATION_PLANS_CONTENT_ID)
     FROM ABC_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.ENTERPRISE_REMEDIATION_PLANS)
        AS SOURCE_DISTINCT_PLANS,
    COUNT(*) AS REPORTING_ROWS,
    COUNT(DISTINCT REMEDIATION_PLAN_KEY) AS REPORT_DISTINCT_PLANS,
    COUNT(*) - COUNT(REMEDIATION_PLAN_KEY) AS NULL_PLAN_KEYS,
    COUNT(DISTINCT CASE WHEN UPPER(TRIM(STATUS)) IN ('IN PROGRESS', 'NOT STARTED')
                       THEN REMEDIATION_PLAN_KEY END) AS IN_PROGRESS_OR_NOT_STARTED,
    COUNT(DISTINCT CASE WHEN UPPER(TRIM(STATUS)) = 'IN PROGRESS'
                       THEN REMEDIATION_PLAN_KEY END) AS IN_PROGRESS,
    COUNT(DISTINCT CASE WHEN UPPER(TRIM(STATUS)) = 'NOT STARTED'
                       THEN REMEDIATION_PLAN_KEY END) AS NOT_STARTED,
    (SELECT COUNT(*) FROM (
        SELECT ARCHER_CONTENT_REMEDIATION_PLANS_CONTENT_ID AS K
        FROM ABC_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.ENTERPRISE_REMEDIATION_PLANS
        EXCEPT SELECT REMEDIATION_PLAN_KEY FROM REPORTING_ROWS_TO_TEST
    )) AS MISSING_PLAN_KEYS,
    (SELECT COUNT(*) FROM (
        SELECT REMEDIATION_PLAN_KEY AS K FROM REPORTING_ROWS_TO_TEST
        EXCEPT SELECT ARCHER_CONTENT_REMEDIATION_PLANS_CONTENT_ID
        FROM ABC_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.ENTERPRISE_REMEDIATION_PLANS
    )) AS EXTRA_PLAN_KEYS,
    COUNT(DISTINCT CASE WHEN RP_BUSINESS_LOOKUP_STATUS NOT IN ('RESOLVED','NOT_RECORDED')
                       THEN REMEDIATION_PLAN_KEY END) AS BU_LOOKUP_ISSUE_PLANS,
    COUNT(DISTINCT CASE WHEN RP_SBU_LOOKUP_STATUS NOT IN ('RESOLVED','NOT_RECORDED')
                       THEN REMEDIATION_PLAN_KEY END) AS SBU_LOOKUP_ISSUE_PLANS,
    COUNT(DISTINCT CASE WHEN RP_ENTITY_LOOKUP_STATUS NOT IN ('RESOLVED','NOT_RECORDED')
                       THEN REMEDIATION_PLAN_KEY END) AS ENTITY_LOOKUP_ISSUE_PLANS
FROM REPORTING_ROWS_TO_TEST;
