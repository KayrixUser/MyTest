
CREATE TASK RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.TASK_C2S_SOX_FINANCE_TREND_HISTORY_WEEKLY
    WAREHOUSE = RTX_ES_ESC_GRC_ETL_PROD_WH
    SCHEDULE = 'USING CRON 0 12 * * 4 America/New_York'
    AUTOCOMMIT = TRUE
    COMMENT = 'NACC-1413: weekly SOX Finance open-finding snapshot; Thursday noon Eastern; retain suspended until upstream coordination and failure notifications are ready.'
AS
INSERT INTO RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.C2S_SOX_FINANCE_TREND_HISTORY
(
    FINDING_ID,
    BUSINESS_UNIT_NAME,
    ORGANIZATION_NAME,
    ENTITY_NAME,
    ENTITY,
    ENTITY_TIER,
    CONTROL_LIBRARY,
    CYCLE,
    SUBCYCLE,
    CONTROL_ID,
    REPORTING_YEAR,
    FISCAL_YEAR,
    FINDING_TITLE,
    FINDING,
    OVERALL_STATUS,
    FINDING_SOURCE,
    FINDING_TYPE,
    CRITICALITY,
    RECORD_CREATOR,
    ASSIGNED_TO,
    REVIEWER,
    SUBMIT_DATE,
    REVIEW_DATE,
    COMMITTED_CLOSURE_DATE,
    REVISED_CLOSURE_DATE,
    DATE_CREATED,
    DATE_CLOSED,
    KEY_CONTROL,
    RTX_SOX_SCOPE,
    TRACKING_ID,
    INSERT_DATE,
    UPDATE_DATE,
    CHECKSUM_VALUE,
    CAPTURE_DATE,
    CAPTURED_AT_UTC
)
WITH CAPTURE_CLOCK AS
(
    SELECT SYSDATE() AS CAPTURED_AT_UTC
),
CAPTURE_CONTEXT AS
(
    SELECT
        CAPTURED_AT_UTC,
        TO_DATE(
            CONVERT_TIMEZONE('UTC', 'America/New_York', CAPTURED_AT_UTC)
        ) AS CAPTURE_DATE
    FROM CAPTURE_CLOCK
),
CAPTURE_INTERVAL AS
(
    SELECT
        CAPTURED_AT_UTC,
        CAPTURE_DATE,
        PREVIOUS_DAY(DATEADD('day', 1, CAPTURE_DATE), 'THU')
            AS CAPTURE_WEEK_START
    FROM CAPTURE_CONTEXT
)
SELECT
    S.FINDING_ID,
    S.BUSINESS_UNIT_NAME,
    S.ORGANIZATION_NAME,
    S.ENTITY_NAME,
    S.ENTITY,
    S.ENTITY_TIER,
    S.CONTROL_LIBRARY,
    S.CYCLE,
    S.SUBCYCLE,
    S.CONTROL_ID,
    S.REPORTING_YEAR,
    S.FISCAL_YEAR,
    S.FINDING_TITLE,
    S.FINDING,
    S.OVERALL_STATUS,
    S.FINDING_SOURCE,
    S.FINDING_TYPE,
    S.CRITICALITY,
    S.RECORD_CREATOR,
    S.ASSIGNED_TO,
    S.REVIEWER,
    S.SUBMIT_DATE,
    S.REVIEW_DATE,
    S.COMMITTED_CLOSURE_DATE,
    S.REVISED_CLOSURE_DATE,
    S.DATE_CREATED,
    S.DATE_CLOSED,
    S.KEY_CONTROL,
    S.RTX_SOX_SCOPE,
    S.TRACKING_ID,
    S.INSERT_DATE,
    S.UPDATE_DATE,
    S.CHECKSUM_VALUE,
    C.CAPTURE_DATE,
    C.CAPTURED_AT_UTC
FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHERSOX_FIN_SOX_DT_FINDINGS_CMS AS S
CROSS JOIN CAPTURE_INTERVAL AS C
WHERE LOWER(TRIM(S.OVERALL_STATUS)) = 'in process'
  AND NOT EXISTS
(
    SELECT 1
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.C2S_SOX_FINANCE_TREND_HISTORY AS H
    WHERE H.CAPTURE_DATE >= C.CAPTURE_WEEK_START
      AND H.CAPTURE_DATE < DATEADD('day', 7, C.CAPTURE_WEEK_START)
);
