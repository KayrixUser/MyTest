WITH REPORT_SOURCE AS (
    SELECT
        V.FINDING_KEY,
        NULLIF(TRIM(F.POLICY_LEVEL_3), '')
            AS DIRECT_POLICY_IDS,
        NULLIF(TRIM(F.CONTROL_STANDARDS), '')
            AS CONTROL_STANDARD_IDS

    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.C2S_ENTERPRISE_FINDINGS_DATA V

    LEFT JOIN RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_FINDINGS F
        ON V.FINDING_KEY
           = F.ARCHER_CONTENT_FINDINGS_CONTENT_ID

    WHERE V.OVERALL_STATUS = 'Open'
)

SELECT
    COUNT(DISTINCT FINDING_KEY)
        AS DASHBOARD_OPEN_FINDINGS,

    COUNT(DISTINCT CASE
        WHEN DIRECT_POLICY_IDS IS NOT NULL
        THEN FINDING_KEY
    END) AS FINDINGS_WITH_DIRECT_POLICY_VALUE,

    COUNT(DISTINCT CASE
        WHEN DIRECT_POLICY_IDS IS NOT NULL
         AND CONTROL_STANDARD_IDS IS NULL
        THEN FINDING_KEY
    END) AS DIRECT_POLICY_WITHOUT_CONTROL_STANDARD

FROM REPORT_SOURCE;
