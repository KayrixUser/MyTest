
WITH POLICY_CHECK AS (
    SELECT
        V.POLICY_LEVEL_3_KEY,

        CASE
            WHEN P.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3_CONTENT_ID
                 IS NULL
            THEN 'NO MATCH'
            ELSE 'MATCHED'
        END AS POLICY_MATCH,

        CASE
            WHEN NULLIF(TRIM(V.POLICY_LEVEL_3_REFERENCE), '') IS NULL
            THEN 'MISSING'
            ELSE 'PRESENT'
        END AS REFERENCE_STATUS,

        CASE
            WHEN NULLIF(TRIM(V.POLICY_LEVEL_3_TITLE), '') IS NULL
            THEN 'MISSING'
            ELSE 'PRESENT'
        END AS TITLE_STATUS

    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.C2S_ENTERPRISE_FINDINGS_POLICY_DATA V

    LEFT JOIN RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3 P
        ON V.POLICY_LEVEL_3_KEY
           = P.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3_CONTENT_ID
)

SELECT
    POLICY_MATCH,
    REFERENCE_STATUS,
    TITLE_STATUS,
    COUNT(*) AS MAPPING_ROWS,
    MIN(POLICY_LEVEL_3_KEY) AS EXAMPLE_POLICY_KEY

FROM POLICY_CHECK

GROUP BY
    POLICY_MATCH,
    REFERENCE_STATUS,
    TITLE_STATUS

ORDER BY MAPPING_ROWS DESC;
