WITH BASE AS (
    SELECT
        CAPTURE_DATE,
        CAPTURED_AT_UTC,
        BUSINESS_UNIT_NAME,
        FINDING_ID,
        COMMITTED_CLOSURE_DATE,
        DATEDIFF(
            'day', COMMITTED_CLOSURE_DATE, CAPTURE_DATE
        ) AS DAYS_PAST_DUE,
        YEAR(COMMITTED_CLOSURE_DATE) AS DUE_YEAR,
        YEAR(CAPTURE_DATE) AS CAPTURE_YEAR
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.C2S_SOX_FINANCE_TREND_HISTORY
    WHERE LOWER(TRIM(OVERALL_STATUS)) = 'in process'
),
RULES AS (
    SELECT
        *,
        DAYS_PAST_DUE <= 30 AS MET_RULE,
        (
            (DUE_YEAR = CAPTURE_YEAR AND DAYS_PAST_DUE > 30)
            OR DUE_YEAR <> CAPTURE_YEAR
        ) AS RISK_RULE
    FROM BASE
)
SELECT
    CAPTURE_DATE,
    CAPTURED_AT_UTC,
    BUSINESS_UNIT_NAME,

    COUNT(DISTINCT FINDING_ID)
        AS TOTAL_IN_PROCESS_FINDINGS,

    COUNT(DISTINCT CASE WHEN MET_RULE THEN FINDING_ID END)
        AS TOTAL_FINDINGS_TARGET_MET,

    COUNT(DISTINCT CASE WHEN RISK_RULE THEN FINDING_ID END)
        AS TOTAL_FINDINGS_TARGET_AT_RISK,

    COUNT(DISTINCT CASE
        WHEN COMMITTED_CLOSURE_DATE IS NULL THEN FINDING_ID
    END) AS MISSING_CLOSURE_DATE,

    COUNT(DISTINCT CASE WHEN MET_RULE THEN FINDING_ID END)
    + COUNT(DISTINCT CASE WHEN RISK_RULE THEN FINDING_ID END)
    - COUNT(DISTINCT CASE
        WHEN MET_RULE OR RISK_RULE THEN FINDING_ID
      END) AS OVERLAPPING_FINDINGS

FROM RULES
GROUP BY
    CAPTURE_DATE,
    CAPTURED_AT_UTC,
    BUSINESS_UNIT_NAME
ORDER BY
    CAPTURE_DATE,
    CAPTURED_AT_UTC,
    BUSINESS_UNIT_NAME;
