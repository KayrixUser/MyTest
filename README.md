
WITH STANDARD_SAMPLE AS (
    SELECT
        C.ARCHER_CONTENT_CONTROL_STANDARDS_CONTENT_ID
            AS CONTROL_STANDARD_KEY,

        C.CONTROL_STANDARD_NAME,

        OBJECT_CONSTRUCT_KEEP_NULL(C.*)
            AS SOURCE_FIELDS

    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_CONTROL_STANDARDS C

    WHERE C.ARCHER_CONTENT_CONTROL_STANDARDS_CONTENT_ID
          IN (7250632, 7250633, 7308452)
)

SELECT
    C.CONTROL_STANDARD_KEY,
    C.CONTROL_STANDARD_NAME,
    F.KEY::VARCHAR AS SOURCE_FIELD,
    F.VALUE::VARCHAR AS SOURCE_VALUE

FROM STANDARD_SAMPLE C,
     LATERAL FLATTEN(INPUT => C.SOURCE_FIELDS) F

WHERE F.KEY::VARCHAR ILIKE '%POLIC%'
   OR F.KEY::VARCHAR = 'CONTROL_STANDARD_NAME'

ORDER BY
    C.CONTROL_STANDARD_KEY,
    SOURCE_FIELD;
