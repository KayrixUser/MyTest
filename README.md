
WITH SOURCE_REFERENCES AS (
    SELECT DISTINCT
        F.ARCHER_CONTENT_FINDINGS_CONTENT_ID AS FINDING_KEY,
        TRY_TO_NUMBER(TRIM(S.VALUE)) AS POLICY_LEVEL_3_KEY

    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_FINDINGS F,
         LATERAL SPLIT_TO_TABLE(
             REPLACE(
                 COALESCE(F.POLICIES_CALCULATED_LEVEL_3, ''),
                 ';',
                 ','
             ),
             ','
         ) S

    WHERE NULLIF(TRIM(S.VALUE), '') IS NOT NULL
),

EXPECTED_ROWS AS (
    SELECT
        R.FINDING_KEY,
        R.POLICY_LEVEL_3_KEY,
        P.POLICY_LEVEL_3_NAME AS POLICY_LEVEL_3_REFERENCE,
        P.POLICY_NAME AS POLICY_LEVEL_3_TITLE

    FROM SOURCE_REFERENCES R

    LEFT JOIN RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3 P
        ON R.POLICY_LEVEL_3_KEY
           = P.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3_CONTENT_ID
),

VIEW_ROWS AS (
    SELECT
        FINDING_KEY,
        POLICY_LEVEL_3_KEY,
        POLICY_LEVEL_3_REFERENCE,
        POLICY_LEVEL_3_TITLE

    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.C2S_ENTERPRISE_FINDINGS_POLICY_DATA
),

MISSING_OR_CHANGED AS (
    SELECT * FROM EXPECTED_ROWS
    MINUS
    SELECT * FROM VIEW_ROWS
),

EXTRA_OR_CHANGED AS (
    SELECT * FROM VIEW_ROWS
    MINUS
    SELECT * FROM EXPECTED_ROWS
),

DUPLICATE_PAIRS AS (
    SELECT
        FINDING_KEY,
        POLICY_LEVEL_3_KEY

    FROM VIEW_ROWS
    GROUP BY FINDING_KEY, POLICY_LEVEL_3_KEY
    HAVING COUNT(*) > 1
),

OPEN_FINDINGS AS (
    SELECT DISTINCT
        ARCHER_CONTENT_FINDINGS_CONTENT_ID AS FINDING_KEY

    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_FINDINGS
    WHERE OVERALL_STATUS = 'Open'
)

SELECT
    '01 Expected source rows' AS CHECK_NAME,
    COUNT(*) AS RESULT
FROM EXPECTED_ROWS

UNION ALL
SELECT '02 Published view rows', COUNT(*)
FROM VIEW_ROWS

UNION ALL
SELECT '03 Missing or changed rows', COUNT(*)
FROM MISSING_OR_CHANGED

UNION ALL
SELECT '04 Extra or changed rows', COUNT(*)
FROM EXTRA_OR_CHANGED

UNION ALL
SELECT '05 Duplicate Finding-Policy pairs', COUNT(*)
FROM DUPLICATE_PAIRS

UNION ALL
SELECT '06 References that did not convert to numeric keys', COUNT(*)
FROM SOURCE_REFERENCES
WHERE POLICY_LEVEL_3_KEY IS NULL

UNION ALL
SELECT '07 Rows with missing policy references or titles', COUNT(*)
FROM EXPECTED_ROWS
WHERE NULLIF(TRIM(POLICY_LEVEL_3_REFERENCE), '') IS NULL
   OR NULLIF(TRIM(POLICY_LEVEL_3_TITLE), '') IS NULL

UNION ALL
SELECT '08 Total open Findings', COUNT(*)
FROM OPEN_FINDINGS

UNION ALL
SELECT
    '09 Open Findings represented by labelled policies',
    COUNT(DISTINCT V.FINDING_KEY)
FROM VIEW_ROWS V
JOIN OPEN_FINDINGS O
    ON V.FINDING_KEY = O.FINDING_KEY
WHERE V.POLICY_LEVEL_3_KEY IS NOT NULL
  AND NULLIF(TRIM(V.POLICY_LEVEL_3_REFERENCE), '') IS NOT NULL

ORDER BY CHECK_NAME;
