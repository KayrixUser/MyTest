WITH RP_SOURCE AS (
    SELECT
        ARCHER_CONTENT_REMEDIATION_PLANS_CONTENT_ID AS PLAN_KEY,
        REMEDIATION_PLAN_ID AS PLAN_NUMBER,
        STATUS AS PLAN_STATUS,
        ARCHER_CONTENT_REMEDIATION_PLANS_CONFIRMED_IN_ARCHER
            AS CONFIRMED_AT,

        MAX(ARCHER_CONTENT_REMEDIATION_PLANS_CONFIRMED_IN_ARCHER)
            OVER () AS LATEST_CONFIRMED_AT,

        COUNT(*) OVER (
            PARTITION BY ARCHER_CONTENT_REMEDIATION_PLANS_CONTENT_ID
        ) AS ROWS_FOR_KEY

    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_REMEDIATION_PLANS
)

SELECT
    CASE
        WHEN CONFIRMED_AT = LATEST_CONFIRMED_AT THEN 'LATEST'
        ELSE 'OLDER_OR_NULL'
    END AS CONFIRMATION_GROUP,

    CONFIRMED_AT,
    PLAN_STATUS,
    COUNT(*) AS SOURCE_ROWS,
    COUNT(DISTINCT PLAN_KEY) AS DISTINCT_PLANS,
    COUNT(*) - COUNT(PLAN_KEY) AS NULL_KEYS,

    COUNT(DISTINCT CASE
        WHEN ROWS_FOR_KEY > 1 THEN PLAN_KEY
    END) AS KEYS_REPEATED_IN_SOURCE,

    COUNT(CASE
        WHEN PLAN_NUMBER = 2115 THEN 1
    END) AS RP_2115_ROWS

FROM RP_SOURCE
GROUP BY
    LATEST_CONFIRMED_AT,
    CONFIRMED_AT,
    PLAN_STATUS

ORDER BY
    CONFIRMED_AT DESC NULLS LAST,
    PLAN_STATUS;
