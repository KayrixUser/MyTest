/* Resolve Entity references. One lookup row per Exception. */
ENTITY_LOOKUP AS (
    SELECT
        X.EXCEPTION_KEY,

        CASE
            WHEN COUNT(DISTINCT X.ENTITY_CONTENT_ID) = 1
            THEN MAX(TRY_TO_NUMBER(X.ENTITY_CONTENT_ID))
        END AS ENTITY_KEY,

        COUNT(DISTINCT X.ENTITY_CONTENT_ID)
            AS ENTITY_REFERENCE_COUNT,

        LISTAGG(
            DISTINCT COALESCE(
                NULLIF(TRIM(E.ENTITY), ''),
                'Unresolved Entity [' || X.ENTITY_CONTENT_ID || ']'
            ),
            '; '
        ) WITHIN GROUP (
            ORDER BY COALESCE(
                NULLIF(TRIM(E.ENTITY), ''),
                'Unresolved Entity [' || X.ENTITY_CONTENT_ID || ']'
            )
        ) AS ENTITY_NAME_LIST,

        COUNT(DISTINCT
            CASE
                WHEN NULLIF(TRIM(E.ENTITY), '') IS NULL
                THEN X.ENTITY_CONTENT_ID
            END
        ) AS ENTITY_UNRESOLVED_COUNT

    FROM (
        SELECT DISTINCT
            ER.ARCHER_CONTENT_EXCEPTION_REQUESTS_CONTENT_ID
                AS EXCEPTION_KEY,
            TRIM(S.VALUE) AS ENTITY_CONTENT_ID

        FROM ABC_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_EXCEPTION_REQUESTS ER,
             LATERAL SPLIT_TO_TABLE(
                 REPLACE(COALESCE(ER.ENTITY, ''), ';', ','),
                 ','
             ) S

        WHERE NULLIF(TRIM(S.VALUE), '') IS NOT NULL
    ) X

    LEFT JOIN ABC_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_ENTITY E
        ON TRY_TO_NUMBER(X.ENTITY_CONTENT_ID)
           = E.ARCHER_CONTENT_ENTITY_CONTENT_ID

    GROUP BY X.EXCEPTION_KEY
),



ER.LAST_UPDATED AS RECORD_LAST_UPDATED_AT



ER.LAST_UPDATED AS RECORD_LAST_UPDATED_AT,

/* Exception's own Entity assignments */
NULLIF(TRIM(ER.ENTITY), '') AS ENTITY_CONTENT_IDS,
EL.ENTITY_KEY AS ENTITY_KEY,
COALESCE(EL.ENTITY_NAME_LIST, 'Not recorded') AS ENTITY_NAME_LIST,
COALESCE(EL.ENTITY_REFERENCE_COUNT, 0) AS ENTITY_REFERENCE_COUNT,
COALESCE(EL.ENTITY_UNRESOLVED_COUNT, 0) AS ENTITY_UNRESOLVED_COUNT


LEFT JOIN RELATED_FINDING_LOOKUP RF
    ON ER.ARCHER_CONTENT_EXCEPTION_REQUESTS_CONTENT_ID
       = RF.EXCEPTION_KEY

LEFT JOIN ENTITY_LOOKUP EL
    ON ER.ARCHER_CONTENT_EXCEPTION_REQUESTS_CONTENT_ID
       = EL.EXCEPTION_KEY;

SELECT
    (
        SELECT COUNT(*)
        FROM ABC_ENTERPRISESERVICES.ES_ESC_GRC_CURATED.ARCHER_CONTENT_EXCEPTION_REQUESTS
    ) AS SOURCE_ROWS,

    COUNT(*) AS VIEW_ROWS,
    COUNT(DISTINCT EXCEPTION_KEY) AS DISTINCT_EXCEPTIONS,
    COUNT(*) - COUNT(EXCEPTION_KEY) AS NULL_EXCEPTION_KEYS,

    COUNT(CASE
        WHEN ENTITY_REFERENCE_COUNT > 1 THEN 1
    END) AS MULTI_ENTITY_EXCEPTIONS,

    COUNT(CASE
        WHEN ENTITY_UNRESOLVED_COUNT > 0 THEN 1
    END) AS UNRESOLVED_ENTITY_EXCEPTIONS

FROM ABC_ENTERPRISESERVICES.ES_ESC_GRC_PUBLISHED.C2S_ENTERPRISE_EXCEPTION_REQUESTS_DATA;
