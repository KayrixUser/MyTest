
WITH POLICY_RECORD AS (
    SELECT
        P.ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3_CONTENT_ID
            AS POLICY_KEY,
        P.POLICY_LEVEL_3_NAME AS POLICY_REFERENCE,
        OBJECT_CONSTRUCT_KEEP_NULL(P.*) AS ROW_DATA
    FROM RTX_ENTERPRISESERVICES.ES_ESC_GRC_CURATED
         .ARCHER_CONTENT_POLICIES_POLICY_LEVEL_3 P
    WHERE TRIM(P.POLICY_LEVEL_3_NAME) = 'DT-RM-001'
),
REFERENCE_FIELDS AS (
    SELECT
        P.POLICY_KEY,
        P.POLICY_REFERENCE,
        K.KEY::VARCHAR AS SOURCE_FIELD,
        CASE
            WHEN IS_NULL_VALUE(K.VALUE) THEN NULL
            ELSE K.VALUE::VARCHAR
        END AS SOURCE_VALUE
    FROM POLICY_RECORD P,
         LATERAL FLATTEN(INPUT => P.ROW_DATA) K
    WHERE K.KEY::VARCHAR ILIKE '%FINDING%'
       OR K.KEY::VARCHAR ILIKE '%ISSUE%'
)
SELECT
    F.POLICY_KEY,
    F.POLICY_REFERENCE,
    F.SOURCE_FIELD,
    CASE
        WHEN NULLIF(TRIM(F.SOURCE_VALUE), '') IS NULL
            THEN 'FIELD EMPTY'
        WHEN MAX(
            CASE
                WHEN TRY_TO_NUMBER(TRIM(T.VALUE)) = 8391422
                THEN 1 ELSE 0
            END
        ) = 1
            THEN 'FOUND FINDING 8391422'
        ELSE 'NOT FOUND IN THIS FIELD'
    END AS FINDING_CHECK
FROM REFERENCE_FIELDS F,
     LATERAL SPLIT_TO_TABLE(
         REPLACE(COALESCE(F.SOURCE_VALUE, ''), ',', ';'),
         ';'
     ) T
GROUP BY
    F.POLICY_KEY,
    F.POLICY_REFERENCE,
    F.SOURCE_FIELD,
    F.SOURCE_VALUE
ORDER BY
    F.POLICY_KEY,
    F.SOURCE_FIELD;
